"""
MOBILE ROBOT KINEMATICS - MATHEMATICAL BACKGROUND

1. FORWARD KINEMATICS (Odometry Model):
   Given wheel velocities → Calculate robot pose (x, y, θ)

   For a differential drive robot:
   - Wheel radius: r
   - Distance between wheels: L (wheel base)
   - Left wheel velocity: ω_l
   - Right wheel velocity: ω_r

   Linear velocity of robot: v = r * (ω_r + ω_l) / 2
   Angular velocity of robot: ω = r * (ω_r - ω_l) / L

   State equations:
   ẋ = v * cos(θ)
   ẏ = v * sin(θ)
   θ̇ = ω

2. INVERSE KINEMATICS:
   Given desired robot velocities (v, ω) → Calculate wheel velocities

   ω_r = (2v + ωL) / (2r)
   ω_l = (2v - ωL) / (2r)

3. POSE ERROR AND CONTROL:
   Given current pose (x, y, θ) and target pose (x_g, y_g, θ_g):
   
   Error in robot frame:
   e_x = (x_g - x) * cos(θ) + (y_g - y) * sin(θ)
   e_y = -(x_g - x) * sin(θ) + (y_g - y) * cos(θ)
   e_θ = atan2(y_g - y, x_g - x) - θ
"""

class DifferentialDriveRobot:
    """
    A complete differential drive robot simulator with full kinematics
    """
    def __init__(self, wheel_radius=0.05, wheel_base=0.3, dt=0.1):
        """
        Initialize robot parameters
        
        Args:
            wheel_radius (float): Radius of wheels (m)
            wheel_base (float): Distance between wheels (m)
            dt (float): Time step for simulation (s)
        """
        self.r = wheel_radius  # Wheel radius
        self.L = wheel_base    # Distance between wheels
        self.dt = dt           # Time step
        
        # Robot state: [x, y, theta]
        self.state = np.array([0.0, 0.0, 0.0])
        
        # Wheel velocities (rad/s)
        self.left_wheel_vel = 0.0
        self.right_wheel_vel = 0.0
        
        # Robot velocities (m/s, rad/s)
        self.linear_vel = 0.0
        self.angular_vel = 0.0
        
        # History for plotting
        self.history = {
            'time': [0],
            'x': [0],
            'y': [0],
            'theta': [0],
            'left_wheel': [0],
            'right_wheel': [0],
            'linear_vel': [0],
            'angular_vel': [0]
        }
        
    def forward_kinematics(self, left_wheel_vel, right_wheel_vel):
        """
        FORWARD KINEMATICS:
        Convert wheel velocities to robot velocities
        
        The math:
        v = (v_right + v_left) / 2
        ω = (v_right - v_left) / L
        
        where:
        v_right = r * ω_right (linear velocity of right wheel)
        v_left = r * ω_left (linear velocity of left wheel)
        
        Args:
            left_wheel_vel: Left wheel angular velocity (rad/s)
            right_wheel_vel: Right wheel angular velocity (rad/s)
        
        Returns:
            linear_vel (float): Robot's linear velocity (m/s)
            angular_vel (float): Robot's angular velocity (rad/s)
        """
        # Convert angular wheel velocities to linear velocities
        v_left = self.r * left_wheel_vel
        v_right = self.r * right_wheel_vel
        
        # Calculate robot velocities
        self.linear_vel = (v_right + v_left) / 2.0
        self.angular_vel = (v_right - v_left) / self.L
        
        self.left_wheel_vel = left_wheel_vel
        self.right_wheel_vel = right_wheel_vel
        
        return self.linear_vel, self.angular_vel
    
    def inverse_kinematics(self, linear_vel, angular_vel):
        """
        INVERSE KINEMATICS:
        Convert desired robot velocities to wheel velocities
        
        The math:
        ω_right = (2*v + ω*L) / (2*r)
        ω_left = (2*v - ω*L) / (2*r)
        
        Args:
            linear_vel: Desired linear velocity (m/s)
            angular_vel: Desired angular velocity (rad/s)
        
        Returns:
            left_wheel_vel (float): Left wheel angular velocity (rad/s)
            right_wheel_vel (float): Right wheel angular velocity (rad/s)
        """
        # Calculate required wheel velocities
        left_wheel_vel = (2 * linear_vel - angular_vel * self.L) / (2 * self.r)
        right_wheel_vel = (2 * linear_vel + angular_vel * self.L) / (2 * self.r)
        
        self.linear_vel = linear_vel
        self.angular_vel = angular_vel
        self.left_wheel_vel = left_wheel_vel
        self.right_wheel_vel = right_wheel_vel
        
        return left_wheel_vel, right_wheel_vel
    
    def update_state(self, linear_vel=None, angular_vel=None, 
                     left_wheel_vel=None, right_wheel_vel=None):
        """
        Update robot pose using the kinematic model
        
        Uses Euler integration:
        x_new = x_old + v * cos(θ) * dt
        y_new = y_old + v * sin(θ) * dt
        θ_new = θ_old + ω * dt
        
        This can use either:
        1. Direct velocity commands (linear_vel, angular_vel)
        2. Wheel velocity commands (left_wheel_vel, right_wheel_vel)
        """
        # If wheel velocities provided, convert to robot velocities
        if left_wheel_vel is not None and right_wheel_vel is not None:
            linear_vel, angular_vel = self.forward_kinematics(
                left_wheel_vel, right_wheel_vel)
        elif linear_vel is not None and angular_vel is not None:
            # If robot velocities provided, convert to wheel velocities
            left_wheel_vel, right_wheel_vel = self.inverse_kinematics(
                linear_vel, angular_vel)
            self.left_wheel_vel = left_wheel_vel
            self.right_wheel_vel = right_wheel_vel
        else:
            linear_vel = self.linear_vel
            angular_vel = self.angular_vel
        
        # Extract current state
        x, y, theta = self.state
        
        # Update state using kinematics
        x_new = x + linear_vel * math.cos(theta) * self.dt
        y_new = y + linear_vel * math.sin(theta) * self.dt
        theta_new = theta + angular_vel * self.dt
        
        # Normalize theta to [-π, π]
        theta_new = math.atan2(math.sin(theta_new), math.cos(theta_new))
        
        # Update state
        self.state = np.array([x_new, y_new, theta_new])
        self.linear_vel = linear_vel
        self.angular_vel = angular_vel
        
        # Record history
        self.history['time'].append(len(self.history['time']) * self.dt)
        self.history['x'].append(x_new)
        self.history['y'].append(y_new)
        self.history['theta'].append(theta_new)
        self.history['left_wheel'].append(self.left_wheel_vel)
        self.history['right_wheel'].append(self.right_wheel_vel)
        self.history['linear_vel'].append(linear_vel)
        self.history['angular_vel'].append(angular_vel)
        
        return self.state.copy()
    
    def get_pose(self):
        """Get current robot pose"""
        return self.state.copy()
    
    def get_error_to_target(self, target_x, target_y, target_theta=None):
        """
        Calculate error between current pose and target pose
        
        Error in global frame:
        Δx = x_target - x_current
        Δy = y_target - y_current
        Δθ = atan2(Δy, Δx) - θ_current
        
        Error in robot frame:
        e_x = Δx * cos(θ) + Δy * sin(θ)    (forward error)
        e_y = -Δx * sin(θ) + Δy * cos(θ)  (lateral error)
        e_θ = Δθ                           (angular error)
        
        Returns:
            distance_error: Euclidean distance to target
            angle_error: Angular error to target
            error_vector: [e_x, e_y, e_theta] in robot frame
        """
        x, y, theta = self.state
        
        dx = target_x - x
        dy = target_y - y
        distance_error = math.sqrt(dx**2 + dy**2)
        
        # Angle to target
        target_angle = math.atan2(dy, dx)
        angle_error = target_angle - theta
        
        # Normalize angle error
        angle_error = math.atan2(math.sin(angle_error), math.cos(angle_error))
        
        # Error in robot frame
        e_x = dx * math.cos(theta) + dy * math.sin(theta)
        e_y = -dx * math.sin(theta) + dy * math.cos(theta)
        
        return distance_error, angle_error, np.array([e_x, e_y, angle_error])

class PositionController:
    """
    Controller to move robot to a target position using inverse kinematics
    """
    def __init__(self, k_linear=1.0, k_angular=2.0, max_linear_vel=1.0, 
                 max_angular_vel=2.0):
        """
        Args:
            k_linear: Proportional gain for linear velocity
            k_angular: Proportional gain for angular velocity
            max_linear_vel: Maximum linear velocity (m/s)
            max_angular_vel: Maximum angular velocity (rad/s)
        """
        self.k_linear = k_linear
        self.k_angular = k_angular
        self.max_linear_vel = max_linear_vel
        self.max_angular_vel = max_angular_vel
    
    def compute_control(self, robot, target_x, target_y, 
                        orientation_control=True):
        """
        Compute control velocities to reach target
        
        Control law:
        v = k_linear * distance_error
        ω = k_angular * angle_error
        
        Both are saturated to maximum values.
        """
        # Get error
        distance_error, angle_error, error_vector = \
            robot.get_error_to_target(target_x, target_y)
        
        # Proportional control
        linear_vel = self.k_linear * distance_error
        angular_vel = self.k_angular * angle_error
        
        # Saturate velocities
        linear_vel = np.clip(linear_vel, -self.max_linear_vel, 
                            self.max_linear_vel)
        angular_vel = np.clip(angular_vel, -self.max_angular_vel, 
                             self.max_angular_vel)
        
        # If we're close to target, stop
        if distance_error < 0.01:
            linear_vel = 0.0
            angular_vel = 0.0
        
        return linear_vel, angular_vel, error_vector

def generate_circular_trajectory(radius=2.0, angular_speed=0.2, 
                                 duration=30, dt=0.1):
    """
    Generate a circular trajectory for the robot to follow
    
    Math: Circular motion
    x(t) = R * cos(ω * t)
    y(t) = R * sin(ω * t)
    
    Required velocities for perfect circle:
    v = R * ω  (tangential velocity)
    ω_robot = ω (angular velocity of robot)
    """
    t = np.arange(0, duration, dt)
    x = radius * np.cos(angular_speed * t)
    y = radius * np.sin(angular_speed * t)
    
    # Calculate reference velocities
    vx = -radius * angular_speed * np.sin(angular_speed * t)
    vy = radius * angular_speed * np.cos(angular_speed * t)
    
    # Reference linear and angular velocities
    v_ref = radius * angular_speed  # Constant tangential velocity
    omega_ref = angular_speed       # Constant angular velocity
    
    return t, x, y, vx, vy, v_ref, omega_ref

def generate_figure8_trajectory(a=2.0, b=1.0, duration=30, dt=0.1):
    """
    Generate a figure-8 (lemniscate) trajectory
    
    Parametric equations:
    x(t) = a * sin(ω * t)
    y(t) = b * sin(2 * ω * t)
    """
    omega = 0.3  # Base frequency
    t = np.arange(0, duration, dt)
    x = a * np.sin(omega * t)
    y = b * np.sin(2 * omega * t)
    
    # Velocities (derivatives)
    vx = a * omega * np.cos(omega * t)
    vy = 2 * b * omega * np.cos(2 * omega * t)
    
    # Speed and direction
    speed = np.sqrt(vx**2 + vy**2)
    direction = np.arctan2(vy, vx)
    
    return t, x, y, vx, vy, speed, direction

def simulate_inverse_kinematics():

    print("INVERSE KINEMATICS - POSITION CONTROL")

    controller = PositionController(k_linear=0.5, k_angular=1.5)
    
    # Initial and target positions
    targets = [(2, 2), (-1, 3), (-2, -1), (3, -2)]
    
    plt.figure(figsize=(10, 8))
    
    for target_x, target_y in targets:
        # Run control loop until reached target
        max_steps = 200
        for step in range(max_steps):
            # Get control velocities
            v, omega, error = controller.compute_control(
                robot, target_x, target_y)
            
            # Apply control
            robot.update_state(linear_vel=v, angular_vel=omega)
            
            # Check if reached target
            distance_error, _, _ = robot.get_error_to_target(
                target_x, target_y)
            if distance_error < 0.05:
                print(f"Reached target ({target_x}, {target_y}) "
                      f"in {step} steps")
                break
        
        # Plot path to this target
        plt.plot(robot.history['x'][-step-1:], 
                robot.history['y'][-step-1:], 
                linewidth=2, label=f'To ({target_x}, {target_y})')
    
    # Plot targets

        plt.plot(tx, ty, 'r*', markersize=20)
        plt.text(tx + 0.1, ty + 0.1, f'Target {i+1}')

        path_line.set_data(path_x, path_y)
        
        # Update velocity data
        time_data.append(frame * robot.dt)
        left_vel_data.append(robot.left_wheel_vel)
        right_vel_data.append(robot.right_wheel_vel)
        left_vel_line.set_data(time_data, left_vel_data)
        right_vel_line.set_data(time_data, right_vel_data)
        
        # Check if reached target
        dist_error, _, _ = robot.get_error_to_target(target_x, target_y)
        if dist_error < 0.05:
            print(f"Target reached at t={frame * robot.dt:.1f}s")
            ani.event_source.stop()
        
        return robot_body, direction_arrow, path_line, left_vel_line, right_vel_line
    
    ani = animation.FuncAnimation(fig, update, frames=300, 
                                 interval=100, blit=False, repeat=False)
    plt.tight_layout()

def mathematical_derivations():
    """
    Print detailed mathematical derivations
    """
    print("\n" + "=" * 70)
    print("DETAILED MATHEMATICAL DERIVATIONS")
    print("=" * 70)
    
    print("""
1. FORWARD KINEMATICS DERIVATION
   =============================
   
   Given a differential drive robot:
   - r: wheel radius
   - L: distance between wheels
   - ω_l, ω_r: angular velocities of left and right wheels
   
   Step 1: Linear velocities of each wheel
   v_l = r * ω_l
   v_r = r * ω_r
   
   Step 2: Robot's linear velocity (center point)
   v = (v_r + v_l) / 2 = r * (ω_r + ω_l) / 2
   
   Step 3: Robot's angular velocity
   The difference in wheel velocities causes rotation:
   ω * L = v_r - v_l
   Therefore: ω = (v_r - v_l) / L = r * (ω_r - ω_l) / L
   
   Step 4: Pose update equations
   ẋ = v * cos(θ)
   ẏ = v * sin(θ)
   θ̇ = ω
   
   Step 5: Discrete-time update (Euler integration)
   x_{k+1} = x_k + v * cos(θ_k) * Δt
   y_{k+1} = y_k + v * sin(θ_k) * Δt
   θ_{k+1} = θ_k + ω * Δt

2. INVERSE KINEMATICS DERIVATION
   ==============================
   
   Given desired robot velocities (v, ω), find wheel velocities.
   
   From forward kinematics:
   v = r * (ω_r + ω_l) / 2  ... (1)
   ω = r * (ω_r - ω_l) / L  ... (2)
   
   From (1): ω_r + ω_l = 2v / r
   From (2): ω_r - ω_l = ωL / r
   
   Solving for ω_r and ω_l:
   Adding: 2ω_r = (2v + ωL) / r
   Subtracting: 2ω_l = (2v - ωL) / r
   
   Therefore:
   ω_r = (2v + ωL) / (2r)
   ω_l = (2v - ωL) / (2r)

3. ERROR CALCULATION IN ROBOT FRAME
   ==================================
   
   Given:
   - Current pose: (x, y, θ)
   - Target pose: (x_g, y_g)
   
   Step 1: Error in global frame
   Δx = x_g - x
   Δy = y_g - y
   
   Step 2: Transform to robot frame using rotation matrix
   R(θ) = [ cos(θ)  sin(θ)]
          [-sin(θ)  cos(θ)]
   
   Error in robot frame:
   [e_x]   [ cos(θ)  sin(θ)] [Δx]
   [e_y] = [-sin(θ)  cos(θ)] [Δy]
   
   e_x = Δx * cos(θ) + Δy * sin(θ)    (forward error)
   e_y = -Δx * sin(θ) + Δy * cos(θ)  (lateral error)
   
   Angular error:
   target_angle = atan2(Δy, Δx)
   e_θ = target_angle - θ

4. CONTROL LAW
   ============
   
   Simple proportional control:
   v = k_p * distance_error
   ω = k_θ * angle_error
   
   Where:
   - k_p: linear velocity gain
   - k_θ: angular velocity gain
   - distance_error = sqrt(Δx² + Δy²)
   - angle_error = atan2(Δy, Δx) - θ

5. SPECIAL MOTIONS
   ================
   
   a) Pure Translation (straight line):
      ω_l = ω_r = v / r
      Result: v = rω, ω = 0
   
   b) Pure Rotation (in-place turn):
      ω_l = -ω_r = -ωL/(2r)
      Result: v = 0, ω = r*(ω_r - ω_l)/L
   
   c) Circular Motion (constant radius R):
      Relationship: v = R * ω
      Wheel velocities:
      v_r = v * (1 + L/(2R))
      v_l = v * (1 - L/(2R))
""")

Detailed Explanation of the Differential Equation Function (diffModel)
Let me break down the 3rd section (the differential equation function) in extreme detail. This is the heart of the forward kinematics - it defines how the robot moves.

The Complete Function
python
def diffModel(x, t, timePoints, sC, rC, dphiLArray, dphiRarray):
    # Interpolate wheel velocities at current time
    dphiLt = np.interp(t, timePoints, dphiLArray)
    dphiRt = np.interp(t, timePoints, dphiRarray)
    
    # Differential equations
    dxdt = 0.5 * rC * dphiLt * np.cos(x[2]) + 0.5 * rC * dphiRt * np.cos(x[2])
    dydt = 0.5 * rC * dphiLt * np.sin(x[2]) + 0.5 * rC * dphiRt * np.sin(x[2])
    dthetadt = -(rC/sC) * dphiLt + (rC/sC) * dphiRt
    
    dxdt = [dxdt, dydt, dthetadt]
    return dxdt
Part 1: Function Parameters
python
def diffModel(x, t, timePoints, sC, rC, dphiLArray, dphiRarray):
Parameter	Meaning	Example Value
x	State vector [x_pos, y_pos, theta] - current robot pose	[500, 500, 0] initially
t	Current time in the simulation	Starts at 0, goes to 100
timePoints	Array of all time values	[0, 0.01, 0.02, ..., 100]

rC	Wheel radius	15
dphiLArray	Left wheel velocity array (for all times)	All 2.0 rad/s
dphiRarray	Right wheel velocity array (for all times)	All 1.4 rad/s
Part 2: Time Interpolation
python
dphiLt = np.interp(t, timePoints, dphiLArray)
dphiRt = np.interp(t, timePoints, dphiRarray)
Why is this needed? The ODE solver (odeint) can call the function at any time, not just the time points you defined. If the solver needs the state at t = 5.7324, we need the wheel velocity at exactly that time.

How np.interp Works:
python
# Example: If wheel velocities change over time
timePoints = [0, 1, 2, 3, 4]        # Known time points
dphiLArray = [2, 2.5, 3, 2.5, 2]   # Left wheel speed at those times

# Result: 2.75 (midpoint between 2.5 and 3)
In our case: Since both arrays are constant (2.0 and 1.4), np.interp simply returns 2.0 and 1.4 regardless of t.

Part 3: The Three Differential Equations (THE CORE)
Equation 1: Change in X Position (dx/dt)
python
dxdt = 0.5 * rC * dphiLt * np.cos(x[2]) + 0.5 * rC * dphiRt * np.cos(x[2])
Let's build this step by step:

Step 1: Wheel Linear Velocities
text
v_left  = rC × dphiLt = 15 × 2.0 = 30 units/s  (left wheel ground speed)
v_right = rC × dphiRt = 15 × 1.4 = 21 units/s  (right wheel ground speed)
Step 2: Robot's Average Linear Velocity
text
v_center = (v_left + v_right) / 2 = (30 + 21) / 2 = 25.5 units/s
This is the speed of the robot's center point.

The code expresses this as:

text
0.5 * rC * dphiLt  +  0.5 * rC * dphiRt
= 0.5 * 15 * 2.0   +  0.5 * 15 * 1.4
= 15              +  10.5
= 25.5 units/s
Step 3: Project to X-axis
The robot moves in the direction it's facing (angle θ). The X component is:

text
ẋ = v_center × cos(θ)
Why cosine?

When robot faces right (θ = 0°): cos(0) = 1 → all motion is in X

When robot faces up (θ = 90°): cos(90) = 0 → no X motion

When robot faces left (θ = 180°): cos(180) = -1 → moves in negative X

Complete equation:

text
ẋ = 0.5 * r * ω_L * cos(θ)  +  0.5 * r * ω_R * cos(θ)
Equation 2: Change in Y Position (dy/dt)
python
dydt = 0.5 * rC * dphiLt * np.sin(x[2]) + 0.5 * rC * dphiRt * np.sin(x[2])
Exactly the same as dx/dt, but uses sine instead of cosine:

text
ẏ = v_center × sin(θ)
Why sine?

When robot faces right (θ = 0°): sin(0) = 0 → no Y motion

When robot faces up (θ = 90°): sin(90) = 1 → all motion is in Y

When robot faces down (θ = 270°): sin(270) = -1 → moves in negative Y

Equation 3: Change in Orientation (dθ/dt)
python
dthetadt = -(rC/sC) * dphiLt + (rC/sC) * dphiRt
This is the MOST IMPORTANT equation - it determines rotation!

Understanding the Physics:
Left wheel contribution: -(rC/sC) * dphiLt

text
Left wheel linear velocity: v_left = 15 × 2.0 = 30 units/s
Contribution to rotation: -(30/60) = -0.5 rad/s

Negative sign because: Left wheel moving forward 
→ pushes robot's left side forward
→ robot rotates CLOCKWISE (negative direction)
Right wheel contribution: +(rC/sC) * dphiRt

text
Right wheel linear velocity: v_right = 15 × 1.4 = 21 units/s  
Contribution to rotation: +(21/60) = +0.35 rad/s

Positive sign because: Right wheel moving forward
→ pushes robot's right side forward
→ robot rotates COUNTER-CLOCKWISE (positive direction)
Net rotation:

text
θ̇ = -0.5 + 0.35 = -0.15 rad/s
The robot rotates clockwise at 0.15 rad/s (about 8.6 degrees/second).

Complete Mathematical Picture
Let me show you what happens at different time steps with actual numbers:

At t = 0 seconds:
text
State: x = 500, y = 500, θ = 0 (facing right)

v_left = 15 × 2.0 = 30 units/s
v_right = 15 × 1.4 = 21 units/s

ẋ = 0.5×30×cos(0) + 0.5×21×cos(0) = 15 + 10.5 = 25.5 units/s
ẏ = 0.5×30×sin(0) + 0.5×21×sin(0) = 0 + 0 = 0 units/s
θ̇ = -(15/60)×2.0 + (15/60)×1.4 = -0.5 + 0.35 = -0.15 rad/s

After 0.01s (one time step):
x = 500 + 25.5 × 0.01 = 500.255
y = 500 + 0 × 0.01 = 500.000
θ = 0 + (-0.15) × 0.01 = -0.0015 rad
At t = 10 seconds:
text
State: x ≈ 750, y ≈ 480, θ ≈ -1.5 rad (about -86°, almost facing down)

ẋ = 25.5 × cos(-1.5) ≈ 25.5 × 0.07 ≈ 1.8 units/s  (mostly moving in Y now)
ẏ = 25.5 × sin(-1.5) ≈ 25.5 × (-0.997) ≈ -25.4 units/s  (moving downward)
θ̇ = -0.15 rad/s (still turning clockwise)
Visual Explanation of the Math
text
                    Robot Orientation (θ)
                         ↑
                         |
                    Y velocity
                         |
    Left wheel          |          Right wheel
    (faster: 2.0)       |          (slower: 1.4)
         ↓              |              ↓
    ┌───────────────────┼───────────────────┐
    │                   |                   │
    │    v_left=30      |     v_right=21    │
    │                   |                   │
    └───────────────────┼───────────────────┘
                        |
              ←--- s=60 ---→
                        |
                    X velocity

Because left wheel is faster:
1. Right side of robot moves slower
2. Robot pivots around a point to the right
3. Net result: Robot curves right (clockwise)
4. Path is a circle with radius R = |v/ω| = 25.5/0.15 = 170 units
Why This Way of Writing It?
The code could have been simplified to:

python
v = 0.5 * rC * (dphiLt + dphiRt)  # Linear velocity
dxdt = v * np.cos(x[2])
dydt = v * np.sin(x[2])
dthetadt = (rC/sC) * (dphiRt - dphiLt)  # Angular velocity
But the author kept it expanded to show the individual wheel contributions more clearly. Both forms are mathematically identical.

Physical Interpretation
The three equations together create a nonholonomic constraint:

The robot CAN move forward/backward

The robot CAN rotate

The robot CANNOT move sideways instantaneously

This is why the path is always tangent to the robot's orientation - a defining characteristic of differential drive robots!

class DifferentialDriveRobot:
    """Complete differential drive robot with forward and inverse kinematics"""
    
    def __init__(self, wheel_radius=15, wheel_base=60):
        """
        Initialize robot parameters
        wheel_radius: Radius of wheels (mm/cm/units)
        wheel_base: Distance between wheels (mm/cm/units)
        """
        self.r = wheel_radius
        self.L = wheel_base
        
        # State: [x, y, theta]
        self.state = np.array([0.0, 0.0, 0.0])
        
        # Wheel velocities
        self.left_wheel_vel = 0.0
        self.right_wheel_vel = 0.0
        
        # Robot velocities
        self.linear_vel = 0.0
        self.angular_vel = 0.0
    
    def forward_kinematics(self, left_vel, right_vel):
        """
        FORWARD KINEMATICS: Wheel velocities → Robot velocities
        
        Equations:
        v = (v_r + v_l) / 2 = r * (ω_r + ω_l) / 2
        ω = (v_r - v_l) / L = r * (ω_r - ω_l) / L
        
        Args:
            left_vel: Left wheel angular velocity (rad/s)
            right_vel: Right wheel angular velocity (rad/s)
        
        Returns:
            (linear_vel, angular_vel): Robot velocities
        """
        self.left_wheel_vel = left_vel
        self.right_wheel_vel = right_vel
        
        # Calculate robot velocities
        self.linear_vel = 0.5 * self.r * (right_vel + left_vel)
        self.angular_vel = (self.r / self.L) * (right_vel - left_vel)
        
        return self.linear_vel, self.angular_vel
    
    def inverse_kinematics(self, linear_vel, angular_vel):
        """
        INVERSE KINEMATICS: Robot velocities → Wheel velocities
        
        Equations:
        ω_r = (2v + ωL) / (2r)
        ω_l = (2v - ωL) / (2r)
        
        Args:
            linear_vel: Desired linear velocity (units/s)
            angular_vel: Desired angular velocity (rad/s)
        
        Returns:
            (left_wheel_vel, right_wheel_vel): Wheel angular velocities
        """
        self.linear_vel = linear_vel
        self.angular_vel = angular_vel
        
        # Calculate required wheel velocities
        self.left_wheel_vel = (2 * linear_vel - angular_vel * self.L) / (2 * self.r)
        self.right_wheel_vel = (2 * linear_vel + angular_vel * self.L) / (2 * self.r)
        
        return self.left_wheel_vel, self.right_wheel_vel
    
    def differential_equation(self, state, t, left_vel_func, right_vel_func):
        """
        Differential equations for ODE solver
        
        State derivatives:
        ẋ = (r/2)(ω_L + ω_R)cos(θ)
        ẏ = (r/2)(ω_L + ω_R)sin(θ)
        θ̇ = (r/L)(ω_R - ω_L)
        
        Args:
            state: Current [x, y, theta]
            t: Current time
            left_vel_func: Function t → left wheel velocity
            right_vel_func: Function t → right wheel velocity
        
        Returns:
            [dx/dt, dy/dt, dtheta/dt]
        """
        x, y, theta = state
        
        # Get wheel velocities at current time
        wL = left_vel_func(t)
        wR = right_vel_func(t)
        
        # Differential equations
        dxdt = 0.5 * self.r * (wL + wR) * np.cos(theta)
        dydt = 0.5 * self.r * (wL + wR) * np.sin(theta)
        dthetadt = (self.r / self.L) * (wR - wL)
        
        return [dxdt, dydt, dthetadt]
    
    def simulate(self, left_vel_func, right_vel_func, t_span, t_eval):
        """
        Simulate robot motion using ODE solver
        
        Args:
            left_vel_func: Function t → left wheel velocity
            right_vel_func: Function t → right wheel velocity
            t_span: (t_start, t_end)
            t_eval: Array of time points
        
        Returns:
            solution: Array of [x, y, theta] at each time step
        """
        from scipy.integrate import solve_ivp
        
        solution = solve_ivp(
            self.differential_equation,
            t_span,
            self.state,
            args=(left_vel_func, right_vel_func),
            t_eval=t_eval,
            method='RK45'
        )
        
        return solution.y.T  # Transpose to get [time_steps, 3]

def generate_circular_trajectory(radius=200, angular_speed=0.2, duration=30, dt=0.01):
    """Generate circular trajectory points and velocities"""
    t = np.arange(0, duration, dt)
    
    # Position
    x = radius * np.cos(angular_speed * t)
    y = radius * np.sin(angular_speed * t)
    
    # Velocities (derivatives)
    vx = -radius * angular_speed * np.sin(angular_speed * t)
    vy = radius * angular_speed * np.cos(angular_speed * t)
    
    # Reference velocities for tracking
    v_ref = radius * angular_speed  # Constant tangential velocity
    omega_ref = np.full_like(t, angular_speed)  # Constant angular velocity
    
    return t, x, y, v_ref, omega_ref

def generate_figure8_trajectory(a=200, b=100, duration=30, dt=0.01):
    """Generate figure-8 (lemniscate) trajectory"""
    omega = 0.3
    t = np.arange(0, duration, dt)
    
    x = a * np.sin(omega * t)
    y = b * np.sin(2 * omega * t)
    
    # Velocities
    vx = a * omega * np.cos(omega * t)
    vy = 2 * b * omega * np.cos(2 * omega * t)
    
    # Speed and direction
    speed = np.sqrt(vx**2 + vy**2)
    angular_vel = np.gradient(np.arctan2(vy, vx)) / dt
    
    return t, x, y, speed, angular_vel

def solve_forward_kinematics():
    """
    Solve forward kinematics with constant wheel velocities
    This generates the simulation data file
    """
    print("Solving Forward Kinematics...")
    
    # Robot parameters
    r = 15  # Wheel radius
    L = 60  # Wheel base
    
    # Time settings
    duration = 100  # seconds
    dt = 0.01
    time_vector = np.linspace(0, duration, int(duration/dt))
    
    # Constant wheel velocities (left faster than right → clockwise curve)
    left_vel = 2.0  # rad/s
    right_vel = 1.4  # rad/s
    
    # Create velocity functions
    left_func = lambda t: left_vel
    right_func = lambda t: right_vel
    
    # Create robot and simulate
    robot = DifferentialDriveRobot(wheel_radius=r, wheel_base=L)
    robot.state = np.array([500.0, 500.0, 0.0])  # Start position
    
    # Differential equations for odeint
    def diff_model(state, t):
        x, y, theta = state
        
        # Get wheel velocities
        wL = left_func(t)
        wR = right_func(t)
        
        # Kinematics equations
        dxdt = 0.5 * r * (wL + wR) * np.cos(theta)
        dydt = 0.5 * r * (wL + wR) * np.sin(theta)
        dthetadt = (r / L) * (wR - wL)
        
        return [dxdt, dydt, dthetadt]
    
    # Solve ODE
    solution = odeint(diff_model, robot.state, time_vector)
    
    # Save data
    np.save('simulationData.npy', solution)
    print(f"Simulation data saved. Shape: {solution.shape}")
    print(f"Time points: {len(time_vector)}, Duration: {duration}s")
    
    return solution, time_vector

def simulate_with_control():
    """Simulate robot with position control to target points"""
    
    print("\n" + "="*60)
    print("POSITION CONTROL SIMULATION")
    print("="*60)
    
    # Robot parameters
    robot = DifferentialDriveRobot(wheel_radius=15, wheel_base=60)
    robot.state = np.array([200.0, 200.0, 0.0])
    
    # Control parameters
    k_linear = 30  # Linear velocity gain
    k_angular = 2  # Angular velocity gain
    dt = 0.1  # Time step
    
    # Target positions
    targets = [
        (600, 400),
        (800, 200),
        (400, 600),
        (200, 400)
    ]
    
    # Storage for plotting
    all_paths = []
    current_path = [robot.state[:2].copy()]
    
    plt.figure(figsize=(10, 8))
    
    for target_idx, (target_x, target_y) in enumerate(targets):
        print(f"\nMoving to Target {target_idx + 1}: ({target_x}, {target_y})")
        
        reached = False
        steps = 0
        max_steps = 500
        
        while not reached and steps < max_steps:
            # Calculate errors
            dx = target_x - robot.state[0]
            dy = target_y - robot.state[1]
            distance_error = np.sqrt(dx**2 + dy**2)
            angle_to_target = np.arctan2(dy, dx)
            angle_error = angle_to_target - robot.state[2]
            
            # Normalize angle error
            angle_error = np.arctan2(np.sin(angle_error), np.cos(angle_error))
            
            # Proportional control
            linear_vel = k_linear * distance_error
            angular_vel = k_angular * angle_error
            
            # Limit velocities
            linear_vel = np.clip(linear_vel, -200, 200)
            angular_vel = np.clip(angular_vel, -3, 3)
            
            # If close to target, stop
            if distance_error < 5:
                linear_vel = 0
                angular_vel = 0
                reached = True
            
            # Convert to wheel velocities and update state
            wL, wR = robot.inverse_kinematics(linear_vel, angular_vel)
            
            # Update state using forward kinematics
            x, y, theta = robot.state
            v, omega = robot.forward_kinematics(wL, wR)
            
            # Euler integration
            robot.state[0] += v * np.cos(theta) * dt
            robot.state[1] += v * np.sin(theta) * dt
            robot.state[2] += omega * dt
            
            current_path.append(robot.state[:2].copy())
            steps += 1
        
        all_paths.append(np.array(current_path))
        print(f"  Reached in {steps} steps")
        
        # Plot this path segment
        path_array = np.array(current_path)
        plt.plot(path_array[:, 0], path_array[:, 1], 
                linewidth=2, label=f'To Target {target_idx + 1}')
        
        current_path = [robot.state[:2].copy()]
    
    # Plot targets
    for i, (tx, ty) in enumerate(targets):
        plt.plot(tx, ty, 'r*', markersize=15)
        plt.text(tx + 10, ty + 10, f'T{i+1}', fontsize=12)
    
    # Plot start
    plt.plot(200, 200, 'go', markersize=12, label='Start')
    
    plt.xlabel('X Position')
    plt.ylabel('Y Position')
    plt.title('Position Control - Robot Moving to Targets')
    plt.axis('equal')
    plt.grid(True, alpha=0.3)
    plt.legend()
    plt.savefig('position_control.png', dpi=150)
    plt.show()

def calculate_velocity_profile(path, dt=0.01):
    """Calculate velocity profile from path data"""
    velocities = np.diff(path, axis=0) / dt
    speeds = np.sqrt(velocities[:, 0]**2 + velocities[:, 1]**2)
    
    # Calculate angular velocities
    headings = np.arctan2(velocities[:, 1], velocities[:, 0])
    angular_velocities = np.diff(headings) / dt
    
    return speeds, angular_velocities

def solve_forward_kinematics():
    """Solve forward kinematics and generate trajectory data"""
    
    # Robot parameters
    r = 15  # Wheel radius
    L = 60  # Wheel base
    
    # Time settings
    duration = 100  # seconds
    dt = 0.01
    time_vector = np.linspace(0, duration, int(duration/dt))
    
    # Wheel velocities (left faster = clockwise curve)
    left_vel = 2.0
    right_vel = 1.4
    
    # Initial state
    initial_state = [500, 500, 0]
    
    # Differential equations
    def diff_model(state, t):
        x, y, theta = state
        
        # Kinematics
        dxdt = 0.5 * r * (left_vel + right_vel) * np.cos(theta)
        dydt = 0.5 * r * (left_vel + right_vel) * np.sin(theta)
        dthetadt = (r / L) * (right_vel - left_vel)
        
        return [dxdt, dydt, dthetadt]
    
    # Solve ODE
    solution = odeint(diff_model, initial_state, time_vector)
    
    # Save data
    np.save('simulationData.npy', solution)
    
    return solution, time_vector

def create_animation(solution, time_vector):
    """Create animation using Matplotlib"""
    
    x = solution[:, 0]
    y = solution[:, 1]
    theta = solution[:, 2]
    
    # Create figure
    fig, ax = plt.subplots(figsize=(12, 8))
    
    # Set limits with some padding

    ax.grid(True, alpha=0.3)
    ax.set_xlabel('X Position')
    ax.set_ylabel('Y Position')
    ax.set_title('Differential Drive Robot Animation')
    
    # Robot dimensions
    robot_length = 80
    robot_width = 50
    
    # Initialize robot rectangle
    robot_rect = Rectangle(
        (x[0] - robot_length/2, y[0] - robot_width/2),
        robot_length, robot_width,
        angle=0,
        fill=True,
        color='blue',
        alpha=0.7,
        label='Robot'
    )
    ax.add_patch(robot_rect)
    
    # Direction arrow
    arrow_length = 40
    direction_arrow = ax.arrow(
        x[0], y[0],
        arrow_length * np.cos(theta[0]),
        arrow_length * np.sin(theta[0]),
        head_width=15, head_length=15,
        fc='red', ec='red'
    )
    
    # Trajectory line
    trajectory_line, = ax.plot([], [], 'y-', linewidth=2, alpha=0.7, label='Path')
    trajectory_x = []
    trajectory_y = []
    
    # Start and target markers
    ax.plot(x[0], y[0], 'go', markersize=12, label='Start')
    
    # Speed display
    speed_text = ax.text(0.02, 0.98, '', transform=ax.transAxes, 
                         verticalalignment='top',
                         bbox=dict(boxstyle='round', facecolor='white', alpha=0.7))
    
    ax.legend(loc='upper right')
    
    def init():
        """Initialize animation"""
        robot_rect.set_xy((x[0] - robot_length/2, y[0] - robot_width/2))
        robot_rect.angle = 0
        direction_arrow.set_data(x=0, y=0, dx=0, dy=0)
        trajectory_line.set_data([], [])
        speed_text.set_text('')
        return robot_rect, direction_arrow, trajectory_line, speed_text
    
    def update(frame):
        """Update animation frame"""
        # Update robot position
        robot_rect.set_xy((x[frame] - robot_length/2, 
                          y[frame] - robot_width/2))
        robot_rect.angle = np.degrees(theta[frame])
        
        # Remove old arrow and create new one
        nonlocal direction_arrow
        if direction_arrow in ax.patches:
            direction_arrow.remove()
        
        direction_arrow = ax.arrow(
            x[frame], y[frame],
            arrow_length * np.cos(theta[frame]),
            arrow_length * np.sin(theta[frame]),
            head_width=15, head_length=15,
            fc='red', ec='red'
        )
        
        # Update trajectory
        trajectory_x.append(x[frame])
        trajectory_y.append(y[frame])
        trajectory_line.set_data(trajectory_x, trajectory_y)
        
        # Calculate speed
        if frame > 0:
            dt = time_vector[1] - time_vector[0]
            dx = x[frame] - x[frame-1]
            dy = y[frame] - y[frame-1]
            speed = np.sqrt(dx**2 + dy**2) / dt
        else:
            speed = 0
        
        # Update text
        speed_text.set_text(
            f'Time: {time_vector[frame]:.1f}s\n'
            f'Position: ({x[frame]:.1f}, {y[frame]:.1f})\n'
            f'Speed: {speed:.1f} units/s\n'
            f'Heading: {np.degrees(theta[frame]):.1f}°'
        )
        
        return robot_rect, direction_arrow, trajectory_line, speed_text
    
    # Create animation
    # Use fewer frames for smoother animation (every 5th frame)
    frame_step = 5
    frames = range(0, len(x), frame_step)
    
    anim = FuncAnimation(
        fig, update, frames=frames,
        init_func=init, blit=False,
        interval=50,  # 50ms between frames = 20 FPS
        repeat=False
    )
    
    plt.tight_layout()
    plt.show()
    
    return anim

import numpy as np

class KalmanFilter:
    """Kalman filter for smooth position tracking"""
    
    def __init__(self, dt=0.1):
        self.dt = dt
        self.state = np.zeros(4)
        
        self.F = np.array([
            [1, 0, dt, 0],
            [0, 1, 0, dt],
            [0, 0, 1, 0],
            [0, 0, 0, 1]
        ])
        
        self.H = np.array([
            [1, 0, 0, 0],
            [0, 1, 0, 0]
        ])
        
        self.Q = np.eye(4) * 0.05
        self.R = np.eye(2) * 0.3
        self.P = np.eye(4) * 1.0
        
    def predict(self):
        self.state = self.F @ self.state
        self.P = self.F @ self.P @ self.F.T + self.Q
        return self.state[:2]
    
    def update(self, measurement):
        y = measurement - self.H @ self.state
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        
        self.state = self.state + K @ y
        self.P = (np.eye(4) - K @ self.H) @ self.P
        
        return self.state[:2]
    
    def set_initial_state(self, position, velocity=None):
        self.state[:2] = position
        if velocity is not None:
            self.state[2:] = velocity
        else:
            self.state[2:] = 0

import numpy as np
from scipy.spatial.distance import euclidean

class CollisionAvoidance:
    @staticmethod
    def compute_avoidance_force(robot_pos, robot_vel, other_robots, obstacles,
                                safety_radius=2.0, max_force=15.0):
        avoidance_force = np.zeros(2)
        
        # Robot-robot avoidance
        for other in other_robots:
            diff = np.array(robot_pos) - np.array(other['pos'])
            dist = np.linalg.norm(diff)
            
            if dist < 0.01:
                diff = np.array([0.1, 0.1])
                dist = 0.14
            
            rel_vel = np.array(robot_vel) - np.array(other.get('velocity', [0, 0]))
            
            if dist < safety_radius * 3:
                rel_speed = np.linalg.norm(rel_vel)
                if rel_speed > 0.01:
                    t_closest = -np.dot(diff, rel_vel) / (rel_speed ** 2)
                    t_closest = max(0, t_closest)
                    closest_dist = np.linalg.norm(diff + rel_vel * t_closest)
                    
                    if closest_dist < safety_radius:
                        if dist > 0:
                            away_dir = diff / dist
                        else:
                            away_dir = np.array([1.0, 0.0])
                        
                        if dist < safety_radius:
                            force_mag = max_force * 2
                        elif closest_dist < safety_radius * 0.5:
                            force_mag = max_force * (1 - closest_dist / safety_radius)
                        else:
                            force_mag = max_force * 0.3 * (1 - closest_dist / safety_radius)
                        
                        perp_dir = np.array([-diff[1], diff[0]])
                        if dist > 0:
                            perp_dir = perp_dir / dist
                        
                        avoidance_force += away_dir * force_mag * 0.7
                        
                        perp_component = np.dot(rel_vel, perp_dir)
                        if abs(perp_component) < 0.5:
                            perp_force = perp_dir * force_mag * 0.3
                            avoidance_force += perp_force
                
                if dist < safety_radius:
                    repulsion = (diff / dist) * max_force * (safety_radius - dist) / safety_radius
                    avoidance_force += repulsion * 3
        
        # Robot-obstacle avoidance
        for obs in obstacles:
            if obs.get('moving', False):
                obs_pos = obs['pos']
                obs_vel = obs.get('velocity', [0, 0])
            else:
                obs_pos = obs['center'] if 'center' in obs else obs['pos']
                obs_vel = [0, 0]
            
            diff = np.array(robot_pos) - np.array(obs_pos)
            dist = np.linalg.norm(diff)
            obs_radius = obs.get('radius', obs.get('size', 1.0))
            safe_dist = obs_radius + safety_radius
            
            if dist < safe_dist * 2:
                if dist < 0.01:
                    diff = np.array([0.1, 0.1])
                    dist = 0.14
                
                rel_vel = np.array(robot_vel) - np.array(obs_vel)
                
                if np.linalg.norm(rel_vel) > 0.01:
                    t_closest = -np.dot(diff, rel_vel) / (np.linalg.norm(rel_vel) ** 2)
                    t_closest = max(0, t_closest)
                    closest_dist = np.linalg.norm(diff + rel_vel * t_closest)
                    
                    if closest_dist < safe_dist:
                        away_dir = diff / dist
                        
                        if dist < safe_dist:
                            force_mag = max_force * 2.5
                        else:
                            force_mag = max_force * (1 - closest_dist / safe_dist) * 1.5
                        
                        avoidance_force += away_dir * force_mag
                        
                        tangent = np.array([-away_dir[1], away_dir[0]])
                        tangent_component = np.dot(rel_vel, tangent)
                        avoidance_force += tangent * tangent_component * 0.3
        
        force_mag = np.linalg.norm(avoidance_force)
        if force_mag > max_force * 3:
            avoidance_force = avoidance_force / force_mag * max_force * 3
        
        return avoidance_force

import cv2
import numpy as np
import time
import threading
import os

class RealTimeArUcoDetector:
    """Real-time ArUco detector that feeds positions to the simulation"""
    
    def __init__(self, camera_id=1, calibration_file=None, workspace_size=(2100, 2850)):
        self.camera_id = camera_id
        self.cap = None
        self.is_camera_open = False
        
        # Workspace
        self.workspace_width_mm, self.workspace_height_mm = workspace_size
        
        # Calibration
        self.camera_matrix = None
        self.distortion_coeffs = None
        self.is_calibrated = False
        
        if calibration_file and os.path.exists(calibration_file):
            self.load_calibration(calibration_file)
        
        # ArUco setup
        try:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5_1000)
        except:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5)
        
        self.parameters = cv2.aruco.DetectorParameters()
        
        # Detection data - store as tuples (not numpy arrays)
        self.robot_positions = {}  # {marker_id: (x_mm, y_mm)}
        self.task_positions = {}   # {marker_id: (x_mm, y_mm)}
        self.start_location = None # (x_mm, y_mm) or None
        self.end_location = None   # (x_mm, y_mm) or None
        self.all_markers = {}      # {marker_id: {'center': (px,py), 'corners': ...}}
        
        # Perspective transform
        self.perspective_matrix = None
        self.is_boundary_set = False
        
        # Threading
        self.running = False
        self.latest_frame = None
        self.detection_thread = None
        
        # Status
        self.frame_count = 0
        self.last_update_time = time.time()
        
        print(f"📷 ArUco Detector: {workspace_size[0]}x{workspace_size[1]}mm")
    
    def load_calibration(self, filename):
        try:
            data = np.load(filename)
            self.camera_matrix = data['camera_matrix']
            self.distortion_coeffs = data['distortion_coeffs']
            self.is_calibrated = True
            print(f"✅ Calibration loaded: {filename}")
            return True
        except Exception as e:
            print(f"❌ Could not load calibration: {e}")
            return False
    
    def open_camera(self):
        self.cap = cv2.VideoCapture(self.camera_id)
        if not self.cap.isOpened():
            print(f"❌ Could not open camera {self.camera_id}")
            return False
        
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1920)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 1080)
        self.is_camera_open = True
        print("✅ Camera opened")
        return True
    
    def close_camera(self):
        if self.cap is not None:
            self.cap.release()
            self.is_camera_open = False
            self.running = False
    
    def undistort_frame(self, frame):
        if not self.is_calibrated or self.camera_matrix is None:
            return frame
        
        h, w = frame.shape[:2]
        new_camera_matrix, roi = cv2.getOptimalNewCameraMatrix(
            self.camera_matrix, self.distortion_coeffs, (w, h), 1, (w, h)
        )
        undistorted = cv2.undistort(
            frame, self.camera_matrix, self.distortion_coeffs, None, new_camera_matrix
        )
        x, y, w, h = roi
        if x > 0 and y > 0 and w > 0 and h > 0:
            undistorted = undistorted[y:y+h, x:x+w]
        return undistorted
    
    def detect_markers(self, frame):
        """Detect all markers and update positions"""
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        
        try:
            corners, ids, rejected = cv2.aruco.detectMarkers(
                gray, self.aruco_dict, parameters=self.parameters
            )
        except:
            corners, ids, rejected = cv2.aruco.detectMarkers(
                gray, self.aruco_dict
            )
        
        # Reset detection data
        self.all_markers = {}
        robot_markers = {}
        task_markers = {}
        corner_markers = {}
        start_pos = None
        end_pos = None
        
        if ids is not None and len(ids) > 0:
            for i in range(len(ids)):
                marker_id = int(ids[i][0])
                marker_corners = corners[i][0]
                
                center_x = int(np.mean(marker_corners[:, 0]))
                center_y = int(np.mean(marker_corners[:, 1]))
                
                self.all_markers[marker_id] = {
                    'center': (center_x, center_y),
                    'corners': marker_corners
                }
                
                # Classify by ID
                if marker_id in [0, 1, 2, 3]:
                    corner_markers[marker_id] = (center_x, center_y)
                elif marker_id in [100, 101, 102]:
                    robot_markers[marker_id] = (center_x, center_y)
                elif marker_id in [20, 21]:
                    task_markers[marker_id] = (center_x, center_y)
                elif marker_id == 10:
                    start_pos = (center_x, center_y)
                elif marker_id == 11:
                    end_pos = (center_x, center_y)
        
        # Setup boundary
        if len(corner_markers) == 4 and not self.is_boundary_set:
            ordered_points = []
            for id in [0, 1, 2, 3]:
                if id in corner_markers:
                    ordered_points.append(corner_markers[id])
            
            if len(ordered_points) == 4:
                src_points = np.array(ordered_points, dtype=np.float32)
                dst_points = np.array([
                    [0, 0],
                    [self.workspace_width_mm, 0],
                    [0, self.workspace_height_mm],
                    [self.workspace_width_mm, self.workspace_height_mm]
                ], dtype=np.float32)
                
                self.perspective_matrix = cv2.getPerspectiveTransform(src_points, dst_points)
                self.is_boundary_set = True
                print("✅ Boundary set!")
        
        # Convert to world coordinates (store as tuples)
        self.robot_positions = {}
        for marker_id, pixel_pos in robot_markers.items():
            if self.perspective_matrix is not None:
                pixel_point = np.array([[[pixel_pos[0], pixel_pos[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.robot_positions[marker_id] = (float(world_x), float(world_y))
            else:
                self.robot_positions[marker_id] = pixel_pos
        
        self.task_positions = {}
        for marker_id, pixel_pos in task_markers.items():
            if self.perspective_matrix is not None:
                pixel_point = np.array([[[pixel_pos[0], pixel_pos[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.task_positions[marker_id] = (float(world_x), float(world_y))
            else:
                self.task_positions[marker_id] = pixel_pos
        
        if start_pos and self.perspective_matrix is not None:
            pixel_point = np.array([[[start_pos[0], start_pos[1]]]], dtype=np.float32)
            world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
            world_x, world_y = world_point[0][0]
            self.start_location = (float(world_x), float(world_y))
        else:
            self.start_location = start_pos
        
        if end_pos and self.perspective_matrix is not None:
            pixel_point = np.array([[[end_pos[0], end_pos[1]]]], dtype=np.float32)
            world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
            world_x, world_y = world_point[0][0]
            self.end_location = (float(world_x), float(world_y))
        else:
            self.end_location = end_pos
        
        self.frame_count += 1
        
        return {
            'robots': self.robot_positions,
            'tasks': self.task_positions,
            'start': self.start_location,
            'end': self.end_location,
            'boundary_set': self.is_boundary_set
        }
    
    def get_positions(self):
        return {
            'robots': self.robot_positions,
            'tasks': self.task_positions,
            'start': self.start_location,
            'end': self.end_location,
            'boundary_set': self.is_boundary_set
        }
    
    def start_detection(self):
        if not self.is_camera_open:
            if not self.open_camera():
                return False
        
        self.running = True
        self.detection_thread = threading.Thread(target=self._detection_loop)
        self.detection_thread.daemon = True
        self.detection_thread.start()
        return True
    
    def _detection_loop(self):
        while self.running:
            ret, frame = self.cap.read()
            if not ret:
                time.sleep(0.01)
                continue
            
            if self.is_calibrated:
                frame = self.undistort_frame(frame)
            
            self.latest_frame = frame
            self.detect_markers(frame)
            time.sleep(0.033)
    
    def stop_detection(self):
        self.running = False
        if self.detection_thread:
            self.detection_thread.join(timeout=1.0)
    
    def show_camera_feed(self):
        """Show camera feed with overlays"""
        while self.running:
            if self.latest_frame is not None:
                frame = self.latest_frame.copy()
                
                # Draw markers
                for marker_id, data in self.all_markers.items():
                    center = data['center']
                    
                    if marker_id in [0, 1, 2, 3]:
                        color = (0, 255, 0)
                    elif marker_id in [100, 101, 102]:
                        color = (0, 0, 255)
                    elif marker_id in [20, 21]:
                        color = (255, 165, 0)
                    elif marker_id == 10:
                        color = (0, 255, 255)
                    elif marker_id == 11:
                        color = (255, 0, 0)
                    else:
                        color = (255, 255, 0)
                    
                    cv2.circle(frame, center, 5, color, -1)
                    cv2.putText(frame, str(marker_id), (center[0]-10, center[1]-10),
                               cv2.FONT_HERSHEY_SIMPLEX, 0.5, color, 2)
                
                # Show positions
                y = 30
                for mid, pos in self.robot_positions.items():
                    cv2.putText(frame, f"R{mid-99}: ({pos[0]:.0f},{pos[1]:.0f})mm",
                               (10, y), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 255), 1)
                    y += 25
                
                if self.start_location:
                    cv2.putText(frame, f"START: ({self.start_location[0]:.0f},{self.start_location[1]:.0f})mm",
                               (10, y), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 0), 1)
                    y += 25
                
                if self.end_location:
                    cv2.putText(frame, f"END: ({self.end_location[0]:.0f},{self.end_location[1]:.0f})mm",
                               (10, y), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 0, 255), 1)
                
                status = "SET" if self.is_boundary_set else "WAITING"
                cv2.putText(frame, f"Boundary: {status}", (10, frame.shape[0]-30),
                           cv2.FONT_HERSHEY_SIMPLEX, 0.6, 
                           (0, 255, 0) if self.is_boundary_set else (0, 0, 255), 2)
                
                cv2.imshow('ArUco Detection', frame)
                if cv2.waitKey(1) & 0xFF == ord('q'):
                    break
            else:
                time.sleep(0.01)
        
        cv2.destroyAllWindows()

from kalman_filter import KalmanFilter

class RobotManager:
    """Manages robots and jobs dynamically"""
    
    def __init__(self, scale=0.01):
        self.scale = scale
        self.robots = []
        self.jobs = []
        self.start_marker = None
        self.end_marker = None
        self.next_job_id = 1
        self.robot_colors = ['#3498DB', '#E74C3C', '#2ECC71', '#F39C12', '#9B59B6']
        self.simulation_time = 0
    
    def update_from_detection(self, positions):
        """Update robots and jobs from detection data"""
        # Update robots
        new_robot_ids = set(positions['robots'].keys())
        current_robot_ids = {r['marker_id'] for r in self.robots}
        
        # Add new robots
        for marker_id in new_robot_ids - current_robot_ids:
            pos = positions['robots'][marker_id]
            sim_x = pos[0] * self.scale
            sim_y = pos[1] * self.scale
            robot_num = marker_id - 99
            
            new_robot = {
                'id': f'R{robot_num}',
                'marker_id': marker_id,
                'start': (sim_x, sim_y),
                'end': (sim_x + 1, sim_y + 1),
                'pos': [sim_x, sim_y],
                'target_pos': [sim_x, sim_y],
                'color': self.robot_colors[len(self.robots) % len(self.robot_colors)],
                'path': [],
                'waypoints': [],
                'current_waypoint': 0,
                'total_distance': 0,
                'assigned_jobs': [],
                'kf': KalmanFilter(dt=0.05),
                'moving': False,
                'velocity': [0, 0],
                'speed': 8.0,
                'avoidance_force': [0, 0],
                'safety_radius': 1.5,
                'real_pos_mm': pos
            }
            new_robot['kf'].set_initial_state([sim_x, sim_y])
            self.robots.append(new_robot)
            print(f"✅ New robot: ID {marker_id} -> R{robot_num}")
        
        # Remove robots
        for robot in self.robots[:]:
            if robot['marker_id'] not in new_robot_ids:
                print(f"❌ Robot {robot['id']} removed")
                self.robots.remove(robot)
        
        # Update positions
        for robot in self.robots:
            if robot['marker_id'] in positions['robots']:
                pos = positions['robots'][robot['marker_id']]
                sim_x = pos[0] * self.scale
                sim_y = pos[1] * self.scale
                robot['real_pos_mm'] = pos
                robot['pos'] = [sim_x, sim_y]
                robot['start'] = (sim_x, sim_y)
                robot['kf'].set_initial_state([sim_x, sim_y])
        
        # Update jobs
        new_job_ids = set(positions['tasks'].keys())
        current_job_ids = {j['marker_id'] for j in self.jobs}
        
        for marker_id in new_job_ids - current_job_ids:
            pos = positions['tasks'][marker_id]
            sim_x = pos[0] * self.scale
            sim_y = pos[1] * self.scale
            
            new_job = {
                'id': f'J{self.next_job_id}',
                'marker_id': marker_id,
                'pos': (sim_x, sim_y),
                'pos_mm': pos,
                'color': self.robot_colors[(self.next_job_id - 1) % len(self.robot_colors)],
                'picked': False,
                'assigned_to': None
            }
            self.jobs.append(new_job)
            self.next_job_id += 1
            print(f"✅ New job: ID {marker_id} -> {new_job['id']}")
        
        for job in self.jobs[:]:
            if job['marker_id'] not in new_job_ids and job['marker_id'] is not None:
                print(f"❌ Job {job['id']} removed")
                self.jobs.remove(job)
        
        for job in self.jobs:
            if job['marker_id'] in positions['tasks']:
                pos = positions['tasks'][job['marker_id']]
                sim_x = pos[0] * self.scale
                sim_y = pos[1] * self.scale
                job['pos'] = (sim_x, sim_y)
                job['pos_mm'] = pos
        
        # Update START
        if positions['start'] is not None:
            pos = positions['start']
            sim_x = pos[0] * self.scale
            sim_y = pos[1] * self.scale
            self.start_marker = (sim_x, sim_y, pos)
        else:
            self.start_marker = None
        
        # Update END
        if positions['end'] is not None:
            pos = positions['end']
            sim_x = pos[0] * self.scale
            sim_y = pos[1] * self.scale
            self.end_marker = (sim_x, sim_y, pos)
        else:
            self.end_marker = None

    def get_jobs(self):

    def get_start(self):

import numpy as np

class SimulationUI:
    """UI and visualization for the simulation"""

        self.sim_height = sim_height

        """Update loop for animation"""
        if not self.animating:
            return []
        
        # Update from detector
        positions = self.detector.get_positions()
        self.robot_manager.update_from_detection(positions)
        
        dt = 0.05
        
        # Update robot positions along paths
        for robot in self.robot_manager.get_robots():
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                robot['moving'] = False
                continue
            
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.3:
                robot['current_waypoint'] += 1
                # Check if reached a job
                for job in robot['assigned_jobs']:
                    if not job['picked'] and np.linalg.norm(np.array(job['pos']) - np.array(robot['pos'])) < 0.5:
                        job['picked'] = True
                        print(f"✅ {robot['id']} picked {job['id']}")
            else:
                robot['moving'] = True
                step = min(robot['speed'] * dt, distance)
                robot['pos'][0] += (direction[0] / distance) * step
                robot['pos'][1] += (direction[1] / distance) * step
        
        # Update UI
        self.ui.draw()
        self.ui.update_status()
        
        return []
    
    def run(self):
        # Start camera feed
        print("\n📷 Starting camera feed...")
        feed_thread = threading.Thread(target=self.detector.show_camera_feed)
        feed_thread.daemon = True
        feed_thread.start()
        
        # Run UI with animation
        self.ui.run_animation(self.update)

            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.3:
                robot['current_waypoint'] += 1
                # Check if reached a job
                for job in robot['assigned_jobs']:
                    if not job['picked'] and np.linalg.norm(np.array(job['pos']) - np.array(robot['pos'])) < 0.5:
                        job['picked'] = True
                        print(f"✅ {robot['id']} picked {job['id']}")
            else:
                robot['moving'] = True
                step = min(robot['speed'] * dt, distance)
                robot['pos'][0] += (direction[0] / distance) * step
                robot['pos'][1] += (direction[1] / distance) * step
        
        # Update UI
        self.ui.draw()
        self.ui.update_status()
        
        return []
    
    def run(self):
        # Start camera feed
        print("\n📷 Starting camera feed...")
        feed_thread = threading.Thread(target=self.detector.show_camera_feed)
        feed_thread.daemon = True
        feed_thread.start()
        
        # Run UI with animation
        self.ui.run_animation(self.update)

import cv2
import numpy as np
import time
import threading
import os

class RealTimeArUcoDetector:
    """Real-time ArUco detector - CLEARS positions each frame"""
    
    def __init__(self, camera_id=1, calibration_file=None, workspace_size=(2100, 2850)):
        self.camera_id = camera_id
        self.cap = None
        self.is_camera_open = False
        
        # Workspace
        self.workspace_width_mm, self.workspace_height_mm = workspace_size
        
        # Calibration
        self.camera_matrix = None
        self.distortion_coeffs = None
        self.is_calibrated = False
        
        if calibration_file and os.path.exists(calibration_file):
            self.load_calibration(calibration_file)
        
        # ArUco setup
        try:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5_1000)
        except:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5)
        
        self.parameters = cv2.aruco.DetectorParameters()
        
        # Detection data - CLEARED each frame
        self.robot_positions = {}      # {marker_id: (x_mm, y_mm)}
        self.task_positions = {}       # {marker_id: (x_mm, y_mm)}
        self.start_location = None     # (x_mm, y_mm) or None
        self.end_location = None       # (x_mm, y_mm) or None
        self.all_markers = {}          # {marker_id: {'center': (px,py), 'corners': ...}}
        self.corner_markers = {}       # {marker_id: (px,py)}
        
        # Perspective transform
        self.perspective_matrix = None
        self.is_boundary_set = False
        
        # Threading
        self.running = False
        self.latest_frame = None
        self.detection_thread = None
        
        # Status
        self.frame_count = 0
        self.last_update_time = time.time()
        
        print(f"📷 ArUco Detector: {workspace_size[0]}x{workspace_size[1]}mm")
        print("   Positions are CLEARED and UPDATED each frame")
    
    def load_calibration(self, filename):
        try:
            data = np.load(filename)
            self.camera_matrix = data['camera_matrix']
            self.distortion_coeffs = data['distortion_coeffs']
            self.is_calibrated = True
            print(f"✅ Calibration loaded: {filename}")
            return True
        except Exception as e:
            print(f"❌ Could not load calibration: {e}")
            return False
    
    def open_camera(self):
        self.cap = cv2.VideoCapture(self.camera_id)
        if not self.cap.isOpened():
            print(f"❌ Could not open camera {self.camera_id}")
            return False
        
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1920)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 1080)
        self.is_camera_open = True
        print("✅ Camera opened")
        return True
    
    def close_camera(self):
        if self.cap is not None:
            self.cap.release()
            self.is_camera_open = False
            self.running = False
    
    def undistort_frame(self, frame):
        if not self.is_calibrated or self.camera_matrix is None:
            return frame
        
        h, w = frame.shape[:2]
        new_camera_matrix, roi = cv2.getOptimalNewCameraMatrix(
            self.camera_matrix, self.distortion_coeffs, (w, h), 1, (w, h)
        )
        undistorted = cv2.undistort(
            frame, self.camera_matrix, self.distortion_coeffs, None, new_camera_matrix
        )
        x, y, w, h = roi
        if x > 0 and y > 0 and w > 0 and h > 0:
            undistorted = undistorted[y:y+h, x:x+w]
        return undistorted
    
    def detect_markers(self, frame):
        """Detect all markers - CLEARS previous positions"""
        
        # ============================================================

    def calculate_paths(self, event):
        robots = self.robot_manager.get_robots()
        jobs = self.robot_manager.get_jobs()
        
        print(f"\n📊 Calculating paths for {len(robots)} robots and {len(jobs)} jobs...")
        
        if not robots:
            print("⚠️ No robots detected! Place robot markers (IDs 100-102)")
            return
        
        if not jobs:
            print("⚠️ No jobs detected! Place job markers (IDs 20-21) or add simulated jobs")
            return
        
        # Assign jobs to robots
        for job in jobs:
            job['assigned_to'] = None
        
        # Simple assignment
        distances = []
        for j_idx, job in enumerate(jobs):
            for r_idx, robot in enumerate(robots):
                dist = ((job['pos'][0] - robot['start'][0])**2 + 
                       (job['pos'][1] - robot['start'][1])**2)**0.5
                distances.append((dist, j_idx, r_idx))
        
        distances.sort()
        assigned = set()
        for dist, j_idx, r_idx in distances:
            if j_idx not in assigned:
                jobs[j_idx]['assigned_to'] = r_idx
                assigned.add(j_idx)
                print(f"   {jobs[j_idx]['id']} -> {robots[r_idx]['id']}")
        
        # Create paths
        for r_idx, robot in enumerate(robots):
            assigned_jobs = [j for j in jobs if j['assigned_to'] == r_idx]
            
            if assigned_jobs:
                path = [robot['start']]
                for job in assigned_jobs:
                    path.append(job['pos'])
                path.append(robot.get('end', robot['start']))
                
                robot['path'] = path
                robot['waypoints'] = path
                robot['assigned_jobs'] = assigned_jobs
                robot['total_distance'] = 0
                for i in range(len(path)-1):
                    robot['total_distance'] += ((path[i+1][0]-path[i][0])**2 + 
                                               (path[i+1][1]-path[i][1])**2)**0.5
                print(f"   {robot['id']}: {len(assigned_jobs)} jobs, {robot['total_distance']:.1f}m")
            else:
                path = [robot['start'], robot.get('end', robot['start'])]
                robot['path'] = path
                robot['waypoints'] = path
                robot['assigned_jobs'] = []
                robot['total_distance'] = ((path[1][0]-path[0][0])**2 +

        self.ui.update_status()
        
        if not self.animating:
            return []
        
        dt = 0.05
        
        # Update robot positions along paths
        for robot in self.robot_manager.get_robots():
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                robot['moving'] = False
                continue
            
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.3:
                robot['current_waypoint'] += 1
                # Check if reached a job
                for job in robot['assigned_jobs']:
                    if not job['picked'] and np.linalg.norm(np.array(job['pos']) - np.array(robot['pos'])) < 0.5:
                        job['picked'] = True
                        print(f"✅ {robot['id']} picked {job['id']}")
            else:
                robot['moving'] = True
                step = min(robot['speed'] * dt, distance)
                robot['pos'][0] += (direction[0] / distance) * step
                robot['pos'][1] += (direction[1] / distance) * step
        
        # Update UI
        self.ui.draw()
        self.ui.update_status()
        
        return []

import cv2
import numpy as np

class RealTimeArUcoDetector:
    """PURE REAL-TIME ArUco detector - NO MAP FALLBACK"""
    
    def __init__(self, camera_id=1, calibration_file=None, workspace_size=(2100, 2850)):
        self.camera_id = camera_id
        self.cap = None
        self.is_camera_open = False
        
        # Workspace
        self.workspace_width_mm, self.workspace_height_mm = workspace_size
        
        # Calibration
        self.camera_matrix = None
        self.distortion_coeffs = None
        self.is_calibrated = False
        
        if calibration_file and os.path.exists(calibration_file):
            self.load_calibration(calibration_file)
        
        # ArUco setup
        try:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5_1000)
        except:
            self.aruco_dict = cv2.aruco.getPredefinedDictionary(cv2.aruco.DICT_5X5)
        
        self.parameters = cv2.aruco.DetectorParameters()
        
        # Detection data - CLEARED each frame
        self.robot_positions = {}      # {marker_id: (x_mm, y_mm)}
        self.task_positions = {}       # {marker_id: (x_mm, y_mm)}
        self.start_location = None     # (x_mm, y_mm) or None
        self.end_location = None       # (x_mm, y_mm) or None
        self.all_markers = {}          # {marker_id: {'center': (px,py), 'corners': ...}}
        self.corner_markers = {}       # {marker_id: (px,py)}
        
        # Perspective transform
        self.perspective_matrix = None
        self.is_boundary_set = False
        
        # Threading
        self.running = False
        self.latest_frame = None
        self.detection_thread = None
        
        # Status
        self.frame_count = 0
        self.detection_count = 0
        
        print(f"📷 PURE REAL-TIME Detector: {workspace_size[0]}x{workspace_size[1]}mm")
        print("   NO MAP FALLBACK - ONLY shows what's currently detected")
    
    def load_calibration(self, filename):
        try:
            data = np.load(filename)
            self.camera_matrix = data['camera_matrix']
            self.distortion_coeffs = data['distortion_coeffs']
            self.is_calibrated = True
            print(f"✅ Calibration loaded: {filename}")
            return True
        except Exception as e:
            print(f"❌ Could not load calibration: {e}")
            return False
    
    def open_camera(self):
        self.cap = cv2.VideoCapture(self.camera_id)
        if not self.cap.isOpened():
            print(f"❌ Could not open camera {self.camera_id}")
            return False
        
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, 1920)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 1080)
        self.is_camera_open = True
        print("✅ Camera opened")
        return True
    
    def close_camera(self):
        if self.cap is not None:
            self.cap.release()
            self.is_camera_open = False
            self.running = False
    
    def undistort_frame(self, frame):
        if not self.is_calibrated or self.camera_matrix is None:
            return frame
        
        h, w = frame.shape[:2]
        new_camera_matrix, roi = cv2.getOptimalNewCameraMatrix(
            self.camera_matrix, self.distortion_coeffs, (w, h), 1, (w, h)
        )
        undistorted = cv2.undistort(
            frame, self.camera_matrix, self.distortion_coeffs, None, new_camera_matrix
        )
        x, y, w, h = roi
        if x > 0 and y > 0 and w > 0 and h > 0:
            undistorted = undistorted[y:y+h, x:x+w]
        return undistorted
    
    def detect_markers(self, frame):
        """Detect all markers - PURE REAL-TIME, CLEAR each frame"""
        
        # ============================================================
        # CLEAR ALL POSITIONS BEFORE DETECTING
        # ============================================================
        self.robot_positions = {}
        self.task_positions = {}
        self.start_location = None
        self.end_location = None
        self.all_markers = {}
        self.corner_markers = {}
        
        gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        
        try:
            corners, ids, rejected = cv2.aruco.detectMarkers(
                gray, self.aruco_dict, parameters=self.parameters
            )
        except:
            corners, ids, rejected = cv2.aruco.detectMarkers(
                gray, self.aruco_dict
            )
        
        # Store raw pixel positions first
        robot_pixels = {}
        task_pixels = {}
        start_pixel = None
        end_pixel = None
        
        if ids is not None and len(ids) > 0:
            for i in range(len(ids)):
                marker_id = int(ids[i][0])
                marker_corners = corners[i][0]
                
                center_x = int(np.mean(marker_corners[:, 0]))
                center_y = int(np.mean(marker_corners[:, 1]))
                
                self.all_markers[marker_id] = {
                    'center': (center_x, center_y),
                    'corners': marker_corners
                }
                
                # Classify by ID
                if marker_id in [0, 1, 2, 3]:
                    self.corner_markers[marker_id] = (center_x, center_y)
                elif marker_id in [100, 101, 102]:
                    robot_pixels[marker_id] = (center_x, center_y)
                elif marker_id in [20, 21]:
                    task_pixels[marker_id] = (center_x, center_y)
                elif marker_id == 10:
                    start_pixel = (center_x, center_y)
                elif marker_id == 11:
                    end_pixel = (center_x, center_y)
        
        # Setup boundary if we have all corners
        if len(self.corner_markers) == 4 and not self.is_boundary_set:
            ordered_points = []
            for id in [0, 1, 2, 3]:
                if id in self.corner_markers:
                    ordered_points.append(self.corner_markers[id])
            
            if len(ordered_points) == 4:
                src_points = np.array(ordered_points, dtype=np.float32)
                dst_points = np.array([
                    [0, 0],
                    [self.workspace_width_mm, 0],
                    [0, self.workspace_height_mm],
                    [self.workspace_width_mm, self.workspace_height_mm]
                ], dtype=np.float32)
                
                self.perspective_matrix = cv2.getPerspectiveTransform(src_points, dst_points)
                self.is_boundary_set = True
                print("✅ Boundary set!")
        
        # ============================================================
        # CONVERT TO WORLD COORDINATES ONLY IF BOUNDARY IS SET
        # ============================================================
        if self.perspective_matrix is not None:
            # Robots
            for marker_id, pixel_pos in robot_pixels.items():
                pixel_point = np.array([[[pixel_pos[0], pixel_pos[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.robot_positions[marker_id] = (float(world_x), float(world_y))
            
            # Tasks
            for marker_id, pixel_pos in task_pixels.items():
                pixel_point = np.array([[[pixel_pos[0], pixel_pos[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.task_positions[marker_id] = (float(world_x), float(world_y))
            
            # START
            if start_pixel is not None:
                pixel_point = np.array([[[start_pixel[0], start_pixel[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.start_location = (float(world_x), float(world_y))
            
            # END
            if end_pixel is not None:
                pixel_point = np.array([[[end_pixel[0], end_pixel[1]]]], dtype=np.float32)
                world_point = cv2.perspectiveTransform(pixel_point, self.perspective_matrix)
                world_x, world_y = world_point[0][0]
                self.end_location = (float(world_x), float(world_y))
        else:
            # No boundary - keep pixel coordinates
            self.robot_positions = robot_pixels
            self.task_positions = task_pixels
            self.start_location = start_pixel
            self.end_location = end_pixel
        
        self.frame_count += 1
        if len(self.all_markers) > 0:
            self.detection_count += 1
        
        return {
            'robots': self.robot_positions,
            'tasks': self.task_positions,
            'start': self.start_location,
            'end': self.end_location,
            'boundary_set': self.is_boundary_set,
            'marker_count': len(self.all_markers)
        }
    
    def get_positions(self):
        """Get current positions - PURE REAL-TIME"""
        return {
            'robots': self.robot_positions,
            'tasks': self.task_positions,
            'start': self.start_location,
            'end': self.end_location,
            'boundary_set': self.is_boundary_set
        }
    
    def start_detection(self):
        if not self.is_camera_open:
            if not self.open_camera():
                return False
        
        self.running = True
        self.detection_thread = threading.Thread(target=self._detection_loop)
        self.detection_thread.daemon = True
        self.detection_thread.start()
        return True
    
    def _detection_loop(self):
        while self.running:
            ret, frame = self.cap.read()
            if not ret:
                time.sleep(0.01)
                continue
            
            if self.is_calibrated:
                frame = self.undistort_frame(frame)
            
            self.latest_frame = frame
            self.detect_markers(frame)
            time.sleep(0.033)
    
    def stop_detection(self):
        self.running = False
        if self.detection_thread:
            self.detection_thread.join(timeout=1.0)
    
    def show_camera_feed(self):

class RealtimeSimulation:
    def __init__(self, calibration_file=None):

    def set_mode(self, mode):
        self.mode = mode
        self.ui.instruction.set_text(f'Mode: {mode} | Click on map')
        self.ui.fig.canvas.draw_idle()
    
    def on_click(self, event):
        if event.inaxes != self.ui.ax:
            return
        
        x, y = event.xdata, event.ydata
        if not (0 <= x <= self.sim_width and 0 <= y <= self.sim_height):
            return
        
        if self.mode == 'add_job':
            x_mm = x / self.scale
            y_mm = y / self.scale
            self.robot_manager.add_simulated_job(x, y, x_mm, y_mm)
            self.ui.draw()
            self.ui.update_status()
    
    def calculate_paths(self, event):
        robots = self.robot_manager.get_robots()
        jobs = self.robot_manager.get_jobs()
        
        print(f"\n📊 Calculating paths for {len(robots)} robots and {len(jobs)} jobs...")
        
        if not robots:
            print("⚠️ No robots detected! Place robot markers (IDs 100-102)")
            return
        
        if not jobs:
            print("⚠️ No jobs detected! Place job markers (IDs 20-21) or add simulated jobs")
            return
        
        # Assign jobs to robots
        for job in jobs:
            job['assigned_to'] = None
        
        distances = []
        for j_idx, job in enumerate(jobs):
            for r_idx, robot in enumerate(robots):
                dist = ((job['pos'][0] - robot['start'][0])**2 + 
                       (job['pos'][1] - robot['start'][1])**2)**0.5
                distances.append((dist, j_idx, r_idx))
        
        distances.sort()
        assigned = set()
        for dist, j_idx, r_idx in distances:
            if j_idx not in assigned:
                jobs[j_idx]['assigned_to'] = r_idx
                assigned.add(j_idx)
                print(f"   {jobs[j_idx]['id']} -> {robots[r_idx]['id']}")
        
        # Create paths
        for r_idx, robot in enumerate(robots):
            assigned_jobs = [j for j in jobs if j['assigned_to'] == r_idx]
            
            if assigned_jobs:
                path = [robot['start']]
                for job in assigned_jobs:
                    path.append(job['pos'])
                path.append(robot.get('end', robot['start']))
                
                robot['path'] = path
                robot['waypoints'] = path
                robot['assigned_jobs'] = assigned_jobs
                robot['total_distance'] = 0
                for i in range(len(path)-1):
                    robot['total_distance'] += ((path[i+1][0]-path[i][0])**2 + 
                                               (path[i+1][1]-path[i][1])**2)**0.5
                print(f"   {robot['id']}: {len(assigned_jobs)} jobs, {robot['total_distance']:.1f}m")
            else:
                path = [robot['start'], robot.get('end', robot['start'])]
                robot['path'] = path
                robot['waypoints'] = path
                robot['assigned_jobs'] = []
                robot['total_distance'] = ((path[1][0]-path[0][0])**2 + 
                                          (path[1][1]-path[0][1])**2)**0.5
        
        self.ui.draw()
        self.ui.update_status()
        print("✅ Paths calculated!")

    def clear_jobs(self, event):

            if job['marker_id'] is None:
                self.robot_manager.jobs.remove(job)
        
        for robot in self.robot_manager.get_robots():
            robot['path'] = []
            robot['waypoints'] = []
            robot['assigned_jobs'] = []
        
        self.ui.draw()
        self.ui.update_status()
        print("🗑️ Simulated jobs cleared")
    
    def update(self, frame):
        """Update loop - PURE REAL-TIME"""
        # Get positions from detector
        positions = self.detector.get_positions()
        
        # Check if anything changed
        current_count = len(positions['robots']) + len(positions['tasks'])
        prev_count = self.last_positions.get('count', 0)
        
        if current_count != prev_count or positions['start'] != self.last_positions.get('start'):
            print(f"🔄 Update: Robots={len(positions['robots'])}, Jobs={len(positions['tasks'])}, START={positions['start'] is not None}, END={positions['end'] is not None}")
        
        self.last_positions = {
            'count': current_count,
            'start': positions['start'],
            'end': positions['end']
        }
        
        # Update robot manager with real-time positions
        self.robot_manager.update_from_detection(positions)
        
        # Force UI update
        self.ui.draw()
        self.ui.update_status()
        
        if not self.animating:
            return []
        
        dt = 0.05
        
        # Update robot positions along paths
        for robot in self.robot_manager.get_robots():
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                robot['moving'] = False
                continue
            
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.3:
                robot['current_waypoint'] += 1
                for job in robot['assigned_jobs']:
                    if not job['picked'] and np.linalg.norm(np.array(job['pos']) - np.array(robot['pos'])) < 0.5:
                        job['picked'] = True
                        print(f"✅ {robot['id']} picked {job['id']}")
            else:
                robot['moving'] = True
                step = min(robot['speed'] * dt, distance)
                robot['pos'][0] += (direction[0] / distance) * step
                robot['pos'][1] += (direction[1] / distance) * step
        
        self.ui.draw()
        self.ui.update_status()
        
        return []
    
    def run(self):
        print("\n📷 Starting camera feed...")
        print("   Press 'q' in the camera window to close it")
        feed_thread = threading.Thread(target=self.detector.show_camera_feed)
        feed_thread.daemon = True
        feed_thread.start()
        
        self.ui.run_animation(self.update)

🐛 KNOWN ISSUES & SOLUTIONS
Issue 1: 8% Distance Error
Problem: Camera calibration has 8% error in distance measurements
Impact: Positions may be off by up to 200mm
Solution:

Re-calibrate with more frames (50-100)

Use more chessboard positions

Ensure perfect lighting

Consider using ChArUco board

Issue 2: Robots/Jobs Not Showing in Simulation
Problem: Markers detected in camera but not in simulation UI
Impact: Simulation not reflecting real world
Solution:

Fix data flow: detector → manager → UI

Ensure positions are cleared each frame

Force UI update on every frame

Issue 3: START/END Overwriting
Problem: Positions overwritten incorrectly when camera moves
Impact: Unreliable position tracking
Solution:

Set start_location = None when not detected

Set end_location = None when not detected

Only update when markers are detected
