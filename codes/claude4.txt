    def assign_jobs_to_robots(self):
        """Assign jobs to nearest robot using greedy algorithm"""
        # Reset assignments
        for job in self.jobs:
            job['assigned_to'] = None
        
        # Calculate distances from each job to each robot
        job_robot_distances = []
        for job_idx, job in enumerate(self.jobs):
            for robot_idx, robot in enumerate(self.robots):
                dist = euclidean(job['pos'], robot['start'])
                job_robot_distances.append((dist, job_idx, robot_idx))
        
        # Sort by distance
        job_robot_distances.sort()
        
        # Assign jobs (greedy)
        assigned_jobs = set()
        robot_job_counts = defaultdict(int)
        max_jobs_per_robot = max(1, len(self.jobs) // len(self.robots) + 1)
        
        for dist, job_idx, robot_idx in job_robot_distances:
            if job_idx not in assigned_jobs and robot_job_counts[robot_idx] < max_jobs_per_robot:
                self.jobs[job_idx]['assigned_to'] = robot_idx
                assigned_jobs.add(job_idx)
                robot_job_counts[robot_idx] += 1
        
        # Assign remaining jobs
        for job_idx in range(len(self.jobs)):
            if job_idx not in assigned_jobs:
                # Assign to robot with fewest jobs
                min_robot = min(range(len(self.robots)), 
                              key=lambda r: robot_job_counts[r])
                self.jobs[job_idx]['assigned_to'] = min_robot
                robot_job_counts[min_robot] += 1
    
    def calculate_optimal_path_for_robot(self, robot_idx):
        """Calculate optimal path for a specific robot"""
        robot = self.robots[robot_idx]
        
        # Get jobs assigned to this robot
        assigned_jobs = [job for job in self.jobs if job['assigned_to'] == robot_idx]
        
        if not assigned_jobs:
            path = [robot['start'], robot['end']]
            distance = euclidean(robot['start'], robot['end'])
            robot['path'] = path
            robot['total_distance'] = distance
            return
        
        # Get job positions
        job_positions = [job['pos'] for job in assigned_jobs]
        
        # Find shortest path (brute force for small number of jobs)
        min_distance = float('inf')
        best_order = None
        
        if len(job_positions) <= 8:  # Brute force for up to 8 jobs
            for perm in itertools.permutations(range(len(job_positions))):
                distance = 0
                current_pos = robot['start']
                
                for job_idx in perm:
                    distance += euclidean(current_pos, job_positions[job_idx])
                    current_pos = job_positions[job_idx]
                
                distance += euclidean(current_pos, robot['end'])
                
                if distance < min_distance:
                    min_distance = distance
                    best_order = perm
        else:
            # Nearest neighbor heuristic for more jobs
            unvisited = list(range(len(job_positions)))
            current_pos = robot['start']
            best_order = []
            total_dist = 0
            
            while unvisited:
                nearest_idx = min(unvisited, 
                                key=lambda i: euclidean(current_pos, job_positions[i]))
                best_order.append(nearest_idx)
                total_dist += euclidean(current_pos, job_positions[nearest_idx])
                current_pos = job_positions[nearest_idx]
                unvisited.remove(nearest_idx)
            
            total_dist += euclidean(current_pos, robot['end'])
            min_distance = total_dist
        
        # Build optimal path
        path = [robot['start']]
        robot['assigned_jobs'] = []
        for job_idx in best_order:
            path.append(job_positions[job_idx])
            robot['assigned_jobs'].append(assigned_jobs[job_idx])
        path.append(robot['end'])
        
        robot['path'] = path
        robot['total_distance'] = min_distance
    
    def calculate_all_paths(self):
        """Calculate optimal paths for all robots"""
        self.assign_jobs_to_robots()
        
        total_distance = 0
        for robot_idx in range(len(self.robots)):
            self.calculate_optimal_path_for_robot(robot_idx)

        return total_distance

    def calculate_optimal_path(self, robot_idx):
        """Calculate optimal path for a robot"""
        robot = self.robots[robot_idx]

            robot['path'] = [robot['start'], robot['end']]
            robot['total_distance'] = euclidean(robot['start'], robot['end'])
            robot['path_points'] = self.generate_curved_path(robot['path'])
            return
        
        job_positions = [job['pos'] for job in assigned_jobs]
        
        min_distance = float('inf')
        best_order = None
        
        if len(job_positions) <= 8:
            for perm in itertools.permutations(range(len(job_positions))):
                distance = 0
                current_pos = robot['start']
                
                for job_idx in perm:
                    distance += euclidean(current_pos, job_positions[job_idx])
                    current_pos = job_positions[job_idx]
                
                distance += euclidean(current_pos, robot['end'])
                
                if distance < min_distance:
                    min_distance = distance
                    best_order = perm
        else:
            unvisited = list(range(len(job_positions)))
            current_pos = robot['start']
            best_order = []
            
            while unvisited:
                nearest_idx = min(unvisited, 
                                key=lambda i: euclidean(current_pos, job_positions[i]))
                best_order.append(nearest_idx)
                current_pos = job_positions[nearest_idx]
                unvisited.remove(nearest_idx)
            
            min_distance = sum(euclidean(robot['start'] if i == 0 else 
                                        job_positions[best_order[i-1]], 
                                        job_positions[best_order[i]]) 
                              for i in range(len(best_order)))
            min_distance += euclidean(job_positions[best_order[-1]], robot['end'])
        
        path = [robot['start']]
        robot['assigned_jobs'] = []
        for job_idx in best_order:
            path.append(job_positions[job_idx])
            robot['assigned_jobs'].append(assigned_jobs[job_idx])
        path.append(robot['end'])
        
        robot['path'] = path
        robot['total_distance'] = min_distance
        robot['path_points'] = self.generate_curved_path(path)
    
    def generate_curved_path(self, waypoints):
        """Generate curved path between waypoints using Bezier curves"""
        if len(waypoints) < 2:
            return waypoints
        
        path_points = []
        num_points_per_segment = 30
        
        for i in range(len(waypoints) - 1):
            p1 = np.array(waypoints[i])
            p2 = np.array(waypoints[i + 1])
            
            # Create control points for curve
            mid = (p1 + p2) / 2
            direction = p2 - p1
            normal = np.array([-direction[1], direction[0]])
            normal = normal / (np.linalg.norm(normal) + 1e-10)
            
            # Add some curvature
            curve_offset = np.random.uniform(-2, 2) * min(5, np.linalg.norm(direction) / 4)
            control_point = mid + normal * curve_offset
            
            # Generate Bezier curve points
            for t in np.linspace(0, 1, num_points_per_segment):
                point = (1-t)**2 * p1 + 2*(1-t)*t * control_point + t**2 * p2
                path_points.append(tuple(point))
        
        path_points.append(waypoints[-1])
        return path_points

        """Create smooth path points between waypoints"""
        if len(waypoints) < 2:

        for i in range(len(waypoints) - 1):
            p1 = np.array(waypoints[i])
            p2 = np.array(waypoints[i + 1])
            
            # Create slight curve using midpoint offset
            mid = (p1 + p2) / 2
            direction = p2 - p1
            normal = np.array([-direction[1], direction[0]])
            norm = np.linalg.norm(normal)
            if norm > 0:
                normal = normal / norm
            
            # Random small offset for natural curve
            offset = np.random.uniform(-1.5, 1.5) * min(3, np.linalg.norm(direction)/5)
            control = mid + normal * offset
            
            # Quadratic Bezier
            for t in np.linspace(0, 1, segments):
                point = (1-t)**2 * p1 + 2*(1-t)*t * control + t**2 * p2
                points.append(tuple(point))
        
        points.append(waypoints[-1])
        return points

    def assign_jobs_to_robots(self):
        """Assign jobs to nearest robot"""
        for job in self.jobs:
            job['assigned_to'] = None
        
        if not self.jobs:
            return
        
        # Calculate distances
        distances = []
        for j_idx, job in enumerate(self.jobs):
            for r_idx, robot in enumerate(self.robots):
                dist = euclidean(job['pos'], robot['start'])
                distances.append((dist, j_idx, r_idx))
        
        distances.sort()
        
        # Greedy assignment
        assigned_jobs = set()
        robot_counts = defaultdict(int)
        max_per_robot = max(1, len(self.jobs) // len(self.robots) + 1)
        
        for dist, j_idx, r_idx in distances:
            if j_idx not in assigned_jobs and robot_counts[r_idx] < max_per_robot:
                self.jobs[j_idx]['assigned_to'] = r_idx
                assigned_jobs.add(j_idx)
                robot_counts[r_idx] += 1
        
        # Assign remaining
        for j_idx in range(len(self.jobs)):
            if j_idx not in assigned_jobs:
                min_robot = min(range(len(self.robots)), 
                              key=lambda r: robot_counts[r])
                self.jobs[j_idx]['assigned_to'] = min_robot
                robot_counts[min_robot] += 1
    
    def calculate_path_for_robot(self, robot_idx):
        """Calculate optimal path for one robot"""
        robot = self.robots[robot_idx]
        assigned = [j for j in self.jobs if j['assigned_to'] == robot_idx]
        
        if not assigned:
            path = [robot['start'], robot['end']]
            robot['path'] = path
            robot['total_distance'] = euclidean(robot['start'], robot['end'])
            robot['assigned_jobs'] = []
            robot['path_points'] = self.create_path_points(path)
            return
        
        positions = [j['pos'] for j in assigned]
        
        # Find shortest path
        min_dist = float('inf')
        best_order = None
        
        if len(positions) <= 8:
            for perm in itertools.permutations(range(len(positions))):
                dist = 0
                curr = robot['start']
                for idx in perm:
                    dist += euclidean(curr, positions[idx])
                    curr = positions[idx]
                dist += euclidean(curr, robot['end'])
                
                if dist < min_dist:
                    min_dist = dist
                    best_order = list(perm)
        else:
            unvisited = list(range(len(positions)))
            curr = robot['start']
            best_order = []
            while unvisited:
                nearest = min(unvisited, key=lambda i: euclidean(curr, positions[i]))
                best_order.append(nearest)
                curr = positions[nearest]
                unvisited.remove(nearest)
        
        # Build path
        path = [robot['start']]
        robot['assigned_jobs'] = []
        for idx in best_order:
            path.append(positions[idx])
            robot['assigned_jobs'].append(assigned[idx])
        path.append(robot['end'])
        
        robot['path'] = path
        robot['total_distance'] = min_dist
        robot['path_points'] = self.create_path_points(path)
    
    def create_path_points(self, waypoints):
        """Create smooth path"""
        if len(waypoints) < 2:
            return [waypoints[0]]
        
        points = []
        segs = 40
        
        for i in range(len(waypoints) - 1):
            p1 = np.array(waypoints[i])
            p2 = np.array(waypoints[i + 1])
            
            mid = (p1 + p2) / 2
            direction = p2 - p1
            normal = np.array([-direction[1], direction[0]])
            norm = np.linalg.norm(normal)
            if norm > 0:
                normal = normal / norm
            
            offset = np.random.uniform(-1.5, 1.5) * min(3, np.linalg.norm(direction)/5)
            control = mid + normal * offset
            
            for t in np.linspace(0, 1, segs):
                point = (1-t)**2 * p1 + 2*(1-t)*t * control + t**2 * p2
                points.append(tuple(point))
        
        points.append(waypoints[-1])
        return points

                robot['current_idx'] = 0

            for job in self.jobs:

                pos = robot['path_points'][robot['current_idx']]
                robot['pos'] = list(pos)

                r_idx = self.robots.index(robot)

class KalmanFilter:
    """Kalman filter for smooth position tracking"""
    def __init__(self, dt=0.1):
        # State: [x, y, vx, vy]
        self.dt = dt
        self.state = np.zeros(4)
        
        # State transition matrix
        self.F = np.array([
            [1, 0, dt, 0],
            [0, 1, 0, dt],
            [0, 0, 1, 0],
            [0, 0, 0, 1]
        ])
        
        # Observation matrix (we observe position only)
        self.H = np.array([
            [1, 0, 0, 0],
            [0, 1, 0, 0]
        ])
        
        # Process noise covariance
        self.Q = np.eye(4) * 0.1
        
        # Measurement noise covariance
        self.R = np.eye(2) * 0.5
        
        # State covariance
        self.P = np.eye(4) * 1.0
        
    def predict(self):
        """Predict next state"""
        self.state = self.F @ self.state
        self.P = self.F @ self.P @ self.F.T + self.Q
        return self.state[:2]
    
    def update(self, measurement):
        """Update state with measurement"""
        y = measurement - self.H @ self.state
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        
        self.state = self.state + K @ y
        self.P = (np.eye(4) - K @ self.H) @ self.P
        
        return self.state[:2]
    
    def set_initial_state(self, position, velocity=None):
        """Set initial state"""
        self.state[:2] = position
        if velocity is not None:
            self.state[2:] = velocity
        else:
            self.state[2:] = 0

    def calculate_shortest_path(self, robot_idx):
        """Calculate shortest straight-line path using TSP"""
        robot = self.robots[robot_idx]
        assigned = [j for j in self.jobs if j['assigned_to'] == robot_idx]
        
        if not assigned:
            path = [robot['start'], robot['end']]
            robot['path'] = path
            robot['waypoints'] = path
            robot['total_distance'] = euclidean(robot['start'], robot['end'])
            robot['assigned_jobs'] = []
            return
        
        positions = [j['pos'] for j in assigned]
        
        # Brute force TSP for shortest path
        min_dist = float('inf')
        best_order = None
        
        if len(positions) <= 8:
            for perm in itertools.permutations(range(len(positions))):
                dist = 0
                curr = robot['start']
                for idx in perm:
                    dist += euclidean(curr, positions[idx])
                    curr = positions[idx]
                dist += euclidean(curr, robot['end'])
                
                if dist < min_dist:
                    min_dist = dist
                    best_order = list(perm)
        else:
            # Nearest neighbor for larger sets
            unvisited = list(range(len(positions)))
            curr = robot['start']
            best_order = []
            while unvisited:
                nearest = min(unvisited, key=lambda i: euclidean(curr, positions[i]))
                best_order.append(nearest)
                curr = positions[nearest]
                unvisited.remove(nearest)
            
            # Calculate actual distance
            min_dist = 0
            curr = robot['start']
            for idx in best_order:
                min_dist += euclidean(curr, positions[idx])
                curr = positions[idx]
            min_dist += euclidean(curr, robot['end'])
        
        # Build path as waypoints (straight lines between points)
        path = [robot['start']]
        robot['assigned_jobs'] = []
        for idx in best_order:
            path.append(positions[idx])
            robot['assigned_jobs'].append(assigned[idx])
        path.append(robot['end'])
        
        robot['path'] = path
        robot['waypoints'] = path
        robot['total_distance'] = min_dist

    def update_animation(self, frame):
        if not self.animating:
            return []
        
        dt = 0.05
        self.simulation_time += dt
        
        all_done = True
        
        for robot in self.robots:
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                continue
            
            all_done = False
            robot['moving'] = True
            
            # Get current target waypoint
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            # Calculate direction and distance to target
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.3:
                # Reached waypoint
                robot['current_waypoint'] += 1
                
                # Check if this was a job
                r_idx = self.robots.index(robot)
                for job in self.jobs:
                    if (not job['picked'] and job['assigned_to'] == r_idx and 
                        euclidean(robot['pos'], job['pos']) < 1.0):
                        job['picked'] = True
                
                if robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                    robot['moving'] = False
                    robot['velocity'] = [0, 0]
            else:
                # Move towards target with Kalman filter
                direction_norm = direction / distance
                velocity = direction_norm * robot['speed']
                
                # Predict with Kalman
                robot['kf'].state[2:] = velocity
                robot['kf'].predict()
                
                # Add small noise to simulate real-world conditions
                noise = np.random.normal(0, 0.05, 2)
                measurement = current + velocity * dt + noise
                
                # Update Kalman filter
                estimated_pos = robot['kf'].update(measurement)
                
                # Update robot position (Kalman filtered)
                robot['pos'] = estimated_pos.tolist()
                robot['velocity'] = velocity.tolist()
                
                # Check if we overshot
                if np.linalg.norm(np.array(robot['pos']) - target) < 0.5:
                    robot['pos'] = target.tolist()
        
        if all_done:
            self.animating = False
            self.action_buttons['anim'].label.set_text('▶️ Start Animation')
            for robot in self.robots:
                robot['moving'] = False
                robot['velocity'] = [0, 0]

class KalmanFilter:
    """Kalman filter for smooth position tracking"""
    def __init__(self, dt=0.1):
        self.dt = dt
        self.state = np.zeros(4)
        
        self.F = np.array([
            [1, 0, dt, 0],
            [0, 1, 0, dt],
            [0, 0, 1, 0],
            [0, 0, 0, 1]
        ])
        
        self.H = np.array([
            [1, 0, 0, 0],
            [0, 1, 0, 0]
        ])
        
        self.Q = np.eye(4) * 0.05
        self.R = np.eye(2) * 0.3
        self.P = np.eye(4) * 1.0
        
    def predict(self):
        self.state = self.F @ self.state
        self.P = self.F @ self.P @ self.F.T + self.Q
        return self.state[:2]
    
    def update(self, measurement):
        y = measurement - self.H @ self.state
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        
        self.state = self.state + K @ y
        self.P = (np.eye(4) - K @ self.H) @ self.P
        
        return self.state[:2]
    
    def set_initial_state(self, position, velocity=None):
        self.state[:2] = position
        if velocity is not None:
            self.state[2:] = velocity
        else:
            self.state[2:] = 0


class CollisionAvoidance:
    """Velocity Obstacle based collision avoidance"""
    
    @staticmethod
    def compute_avoidance_force(robot_pos, robot_vel, other_robots, 
                                safety_radius=2.0, max_force=15.0):
        """
        Compute collision avoidance force using Velocity Obstacle approach
        """
        avoidance_force = np.zeros(2)
        
        for other in other_robots:
            # Vector from other robot to current robot
            diff = np.array(robot_pos) - np.array(other['pos'])
            dist = np.linalg.norm(diff)
            
            if dist < 0.01:  # Avoid division by zero
                diff = np.array([0.1, 0.1])
                dist = 0.14
            
            # Relative velocity
            rel_vel = np.array(robot_vel) - np.array(other.get('velocity', [0, 0]))
            
            # Check if collision course
            if dist < safety_radius * 3:
                # Time to closest approach
                rel_speed = np.linalg.norm(rel_vel)
                if rel_speed > 0.01:
                    t_closest = -np.dot(diff, rel_vel) / (rel_speed ** 2)
                    t_closest = max(0, t_closest)
                    
                    # Closest approach distance
                    closest_dist = np.linalg.norm(diff + rel_vel * t_closest)
                    
                    # Compute avoidance force
                    if closest_dist < safety_radius:
                        # Direction away from other robot
                        if dist > 0:
                            away_dir = diff / dist
                        else:
                            away_dir = np.array([1.0, 0.0])
                        
                        # Force magnitude based on distance
                        if dist < safety_radius:
                            # Emergency braking
                            force_mag = max_force * 2
                        elif closest_dist < safety_radius * 0.5:
                            force_mag = max_force * (1 - closest_dist / safety_radius)
                        else:
                            force_mag = max_force * 0.3 * (1 - closest_dist / safety_radius)
                        
                        # Add perpendicular component for smoother avoidance
                        perp_dir = np.array([-diff[1], diff[0]])
                        if dist > 0:
                            perp_dir = perp_dir / dist
                        
                        # Combine away and perpendicular forces
                        avoidance_force += away_dir * force_mag * 0.7
                        
                        # Add perpendicular force based on relative velocity
                        perp_component = np.dot(rel_vel, perp_dir)
                        if abs(perp_component) < 0.5:
                            # Choose perpendicular direction that gives more clearance
                            perp_force = perp_dir * force_mag * 0.3
                            avoidance_force += perp_force
                
                # Additional repulsion when very close
                if dist < safety_radius:
                    repulsion = (diff / dist) * max_force * (safety_radius - dist) / safety_radius
                    avoidance_force += repulsion * 3
        
        # Limit maximum force
        force_mag = np.linalg.norm(avoidance_force)
        if force_mag > max_force * 2:
            avoidance_force = avoidance_force / force_mag * max_force * 2
        
        return avoidance_force

    def on_click(self, event):
        if event.inaxes != self.ax:
            return
        
        x, y = event.xdata, event.ydata
        if not (0 <= x <= self.boundary_width and 0 <= y <= self.boundary_height):
            return
        
        x, y = round(x, 1), round(y, 1)
        
        if self.mode == 'add_job':
            self.add_job(x, y)
        elif self.mode == 'set_start':
            robot = self.robots[self.selected_robot]
            robot['start'] = (x, y)
            robot['pos'] = [x, y]
            robot['target_pos'] = [x, y]
            robot['kf'].set_initial_state([x, y])
            self.draw_all()
            self.update_status()
        elif self.mode == 'set_end':
            self.robots[self.selected_robot]['end'] = (x, y)
            self.draw_all()
            self.update_status()
        elif self.mode == 'move_job':
            self.handle_move_job(x, y)
    
    def add_job(self, x, y):
        for job in self.jobs:
            if euclidean((x, y), job['pos']) < 1.0:
                return
        
        colors = ['#E74C3C', '#3498DB', '#2ECC71', '#F39C12', '#9B59B6',
                  '#1ABC9C', '#E67E22', '#16A085', '#C0392B', '#8E44AD']
        
        job = {
            'id': f'J{self.next_job_id}',
            'pos': (x, y),
            'color': colors[(self.next_job_id - 1) % len(colors)],
            'picked': False,
            'assigned_to': None
        }
        self.jobs.append(job)
        self.next_job_id += 1
        self.draw_all()
        self.update_status()
    
    def handle_move_job(self, x, y):
        if self.selected_job_idx is None:
            for i, job in enumerate(self.jobs):
                if euclidean((x, y), job['pos']) < 1.0:
                    self.selected_job_idx = i
                    self.instruction.set_text(f'Selected {job["id"]} | Click new location')
                    self.fig.canvas.draw_idle()
                    return
        else:
            self.jobs[self.selected_job_idx]['pos'] = (x, y)
            self.selected_job_idx = None
            self.set_mode('add_job')
            self.draw_all()
            self.update_status()

    def assign_jobs_to_robots(self):
        for job in self.jobs:
            job['assigned_to'] = None
        
        if not self.jobs:
            return
        
        distances = []
        for j_idx, job in enumerate(self.jobs):
            for r_idx, robot in enumerate(self.robots):
                dist = euclidean(job['pos'], robot['start'])
                distances.append((dist, j_idx, r_idx))
        
        distances.sort()
        
        assigned_jobs = set()
        robot_counts = defaultdict(int)
        max_per_robot = max(1, len(self.jobs) // len(self.robots) + 1)
        
        for dist, j_idx, r_idx in distances:
            if j_idx not in assigned_jobs and robot_counts[r_idx] < max_per_robot:
                self.jobs[j_idx]['assigned_to'] = r_idx
                assigned_jobs.add(j_idx)
                robot_counts[r_idx] += 1
        
        for j_idx in range(len(self.jobs)):
            if j_idx not in assigned_jobs:
                min_robot = min(range(len(self.robots)), 
                              key=lambda r: robot_counts[r])
                self.jobs[j_idx]['assigned_to'] = min_robot
                robot_counts[min_robot] += 1
    
    def calculate_shortest_path(self, robot_idx):
        """Calculate shortest straight-line path using TSP"""
        robot = self.robots[robot_idx]
        assigned = [j for j in self.jobs if j['assigned_to'] == robot_idx]
        
        if not assigned:
            path = [robot['start'], robot['end']]
            robot['path'] = path
            robot['waypoints'] = path
            robot['total_distance'] = euclidean(robot['start'], robot['end'])
            robot['assigned_jobs'] = []
            return
        
        positions = [j['pos'] for j in assigned]
        
        min_dist = float('inf')
        best_order = None
        
        if len(positions) <= 8:
            for perm in itertools.permutations(range(len(positions))):
                dist = 0
                curr = robot['start']
                for idx in perm:
                    dist += euclidean(curr, positions[idx])
                    curr = positions[idx]
                dist += euclidean(curr, robot['end'])
                
                if dist < min_dist:
                    min_dist = dist
                    best_order = list(perm)
        else:
            unvisited = list(range(len(positions)))
            curr = robot['start']
            best_order = []
            while unvisited:
                nearest = min(unvisited, key=lambda i: euclidean(curr, positions[i]))
                best_order.append(nearest)
                curr = positions[nearest]
                unvisited.remove(nearest)
            
            min_dist = 0
            curr = robot['start']
            for idx in best_order:
                min_dist += euclidean(curr, positions[idx])
                curr = positions[idx]
            min_dist += euclidean(curr, robot['end'])
        
        path = [robot['start']]
        robot['assigned_jobs'] = []
        for idx in best_order:
            path.append(positions[idx])
            robot['assigned_jobs'].append(assigned[idx])
        path.append(robot['end'])
        
        robot['path'] = path
        robot['waypoints'] = path
        robot['total_distance'] = min_dist
    
    def calculate_all_paths(self, event):
        self.assign_jobs_to_robots()
        for i in range(len(self.robots)):
            self.calculate_shortest_path(i)
        self.draw_all()
        self.update_status()
    
    def check_collisions(self):
        """Check for collisions between robots"""
        collisions = []
        for i in range(len(self.robots)):
            for j in range(i + 1, len(self.robots)):
                dist = euclidean(self.robots[i]['pos'], self.robots[j]['pos'])
                if dist < 1.0:  # Collision threshold
                    collisions.append((i, j, dist))
        return collisions

    def update_animation(self, frame):
        if not self.animating:
            return []
        
        dt = 0.05
        self.simulation_time += dt
        
        all_done = True
        
        # First, compute avoidance forces for all robots
        for robot in self.robots:
            robot['avoidance_force'] = [0, 0]
        
        if self.collision_avoidance_enabled:
            for i, robot in enumerate(self.robots):
                if robot['moving']:
                    other_robots = [self.robots[j] for j in range(len(self.robots)) if j != i]
                    avoidance = CollisionAvoidance.compute_avoidance_force(
                        robot['pos'], robot['velocity'], other_robots,
                        safety_radius=robot['safety_radius']
                    )
                    robot['avoidance_force'] = avoidance.tolist()
        
        # Update each robot
        for robot in self.robots:
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                robot['moving'] = False
                robot['velocity'] = [0, 0]
                continue
            
            all_done = False
            robot['moving'] = True
            
            # Get current target
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.5:
                # Reached waypoint
                robot['current_waypoint'] += 1
                
                r_idx = self.robots.index(robot)
                for job in self.jobs:
                    if (not job['picked'] and job['assigned_to'] == r_idx and 
                        euclidean(robot['pos'], job['pos']) < 1.0):
                        job['picked'] = True
                
                if robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                    robot['moving'] = False
                    robot['velocity'] = [0, 0]
            else:
                # Calculate desired velocity towards target
                direction_norm = direction / distance
                desired_velocity = direction_norm * robot['speed']
                
                # Apply collision avoidance
                if self.collision_avoidance_enabled:
                    avoidance = np.array(robot['avoidance_force'])
                    # Combine desired velocity with avoidance
                    combined_velocity = desired_velocity + avoidance
                    
                    # Limit speed
                    speed = np.linalg.norm(combined_velocity)
                    if speed > robot['speed'] * 1.5:
                        combined_velocity = combined_velocity / speed * robot['speed'] * 1.5
                    if speed < robot['speed'] * 0.3 and distance > 1.0:
                        # Ensure minimum speed towards target
                        combined_velocity = direction_norm * robot['speed'] * 0.5
                else:
                    combined_velocity = desired_velocity
                
                # Update with Kalman filter
                robot['kf'].state[2:] = combined_velocity
                robot['kf'].predict()
                
                # Add noise
                noise = np.random.normal(0, 0.05, 2)
                measurement = current + combined_velocity * dt + noise
                
                # Kalman update
                estimated_pos = robot['kf'].update(measurement)
                
                # Boundary avoidance
                estimated_pos[0] = np.clip(estimated_pos[0], 0.5, self.boundary_width - 0.5)
                estimated_pos[1] = np.clip(estimated_pos[1], 0.5, self.boundary_height - 0.5)
                
                robot['pos'] = estimated_pos.tolist()
                robot['velocity'] = combined_velocity.tolist()
        
        # Check collisions
        collisions = self.check_collisions()
        if collisions:
            self.collision_warning.set_text('⚠️ COLLISION DETECTED!')
            self.collision_warning.set_bbox(dict(boxstyle='round,pad=0.3', 
                                                  facecolor='#FFE5E5', alpha=0.9))
        else:
            self.collision_warning.set_text('')
            self.collision_warning.set_bbox(dict(boxstyle='round,pad=0.3', 
                                                  facecolor='#FFE5E5', alpha=0))
        
        if all_done:
            self.animating = False
            self.action_buttons['anim'].label.set_text('▶️ Start Animation')
            for robot in self.robots:
                robot['moving'] = False
                robot['velocity'] = [0, 0]
        
        self.draw_all()
        self.update_status()
        return []
    
    def draw_all(self):
        # Clear
        for artist in self.ax.collections[:]:

class KalmanFilter:
    """Kalman filter for smooth position tracking"""
    def __init__(self, dt=0.1):
        self.dt = dt
        self.state = np.zeros(4)
        
        self.F = np.array([
            [1, 0, dt, 0],
            [0, 1, 0, dt],
            [0, 0, 1, 0],
            [0, 0, 0, 1]
        ])
        
        self.H = np.array([
            [1, 0, 0, 0],
            [0, 1, 0, 0]
        ])
        
        self.Q = np.eye(4) * 0.05
        self.R = np.eye(2) * 0.3
        self.P = np.eye(4) * 1.0
        
    def predict(self):
        self.state = self.F @ self.state
        self.P = self.F @ self.P @ self.F.T + self.Q
        return self.state[:2]
    
    def update(self, measurement):
        y = measurement - self.H @ self.state
        S = self.H @ self.P @ self.H.T + self.R
        K = self.P @ self.H.T @ np.linalg.inv(S)
        
        self.state = self.state + K @ y
        self.P = (np.eye(4) - K @ self.H) @ self.P
        
        return self.state[:2]
    
    def set_initial_state(self, position, velocity=None):
        self.state[:2] = position
        if velocity is not None:
            self.state[2:] = velocity
        else:
            self.state[2:] = 0


class CollisionAvoidance:
    """Enhanced collision avoidance with obstacle support"""
    
    @staticmethod
    def compute_avoidance_force(robot_pos, robot_vel, other_robots, obstacles,
                                safety_radius=2.0, max_force=15.0):
        avoidance_force = np.zeros(2)
        
        # Robot-robot avoidance
        for other in other_robots:
            diff = np.array(robot_pos) - np.array(other['pos'])
            dist = np.linalg.norm(diff)
            
            if dist < 0.01:
                diff = np.array([0.1, 0.1])
                dist = 0.14
            
            rel_vel = np.array(robot_vel) - np.array(other.get('velocity', [0, 0]))
            
            if dist < safety_radius * 3:
                rel_speed = np.linalg.norm(rel_vel)
                if rel_speed > 0.01:
                    t_closest = -np.dot(diff, rel_vel) / (rel_speed ** 2)
                    t_closest = max(0, t_closest)
                    closest_dist = np.linalg.norm(diff + rel_vel * t_closest)
                    
                    if closest_dist < safety_radius:
                        if dist > 0:
                            away_dir = diff / dist
                        else:
                            away_dir = np.array([1.0, 0.0])
                        
                        if dist < safety_radius:
                            force_mag = max_force * 2
                        elif closest_dist < safety_radius * 0.5:
                            force_mag = max_force * (1 - closest_dist / safety_radius)
                        else:
                            force_mag = max_force * 0.3 * (1 - closest_dist / safety_radius)
                        
                        perp_dir = np.array([-diff[1], diff[0]])
                        if dist > 0:
                            perp_dir = perp_dir / dist
                        
                        avoidance_force += away_dir * force_mag * 0.7
                        
                        perp_component = np.dot(rel_vel, perp_dir)
                        if abs(perp_component) < 0.5:
                            perp_force = perp_dir * force_mag * 0.3
                            avoidance_force += perp_force
                
                if dist < safety_radius:
                    repulsion = (diff / dist) * max_force * (safety_radius - dist) / safety_radius
                    avoidance_force += repulsion * 3
        
        # Robot-obstacle avoidance
        for obs in obstacles:
            if obs.get('moving', False):
                obs_pos = obs['pos']
                obs_vel = obs.get('velocity', [0, 0])
            else:
                obs_pos = obs['center'] if 'center' in obs else obs['pos']
                obs_vel = [0, 0]
            
            diff = np.array(robot_pos) - np.array(obs_pos)
            dist = np.linalg.norm(diff)
            obs_radius = obs.get('radius', obs.get('size', 1.0))
            safe_dist = obs_radius + safety_radius
            
            if dist < safe_dist * 2:
                if dist < 0.01:
                    diff = np.array([0.1, 0.1])
                    dist = 0.14
                
                rel_vel = np.array(robot_vel) - np.array(obs_vel)
                
                if np.linalg.norm(rel_vel) > 0.01:
                    t_closest = -np.dot(diff, rel_vel) / (np.linalg.norm(rel_vel) ** 2)
                    t_closest = max(0, t_closest)
                    closest_dist = np.linalg.norm(diff + rel_vel * t_closest)
                    
                    if closest_dist < safe_dist:
                        away_dir = diff / dist
                        
                        if dist < safe_dist:
                            force_mag = max_force * 2.5
                        else:
                            force_mag = max_force * (1 - closest_dist / safe_dist) * 1.5
                        
                        avoidance_force += away_dir * force_mag
                        
                        # Add tangential component to slide along obstacle
                        tangent = np.array([-away_dir[1], away_dir[0]])
                        tangent_component = np.dot(rel_vel, tangent)
                        avoidance_force += tangent * tangent_component * 0.3
        
        # Limit maximum force
        force_mag = np.linalg.norm(avoidance_force)
        if force_mag > max_force * 3:
            avoidance_force = avoidance_force / force_mag * max_force * 3
        
        return avoidance_force
    
    @staticmethod
    def check_obstacle_collision(pos, obstacles):
        """Check if a point collides with any obstacle"""
        for obs in obstacles:
            if obs.get('moving', False):
                obs_pos = obs['pos']
                obs_radius = obs.get('radius', 1.0)
            else:
                obs_pos = obs['center'] if 'center' in obs else obs['pos']
                obs_radius = obs.get('radius', obs.get('size', 1.0))
            
            dist = euclidean(pos, obs_pos)
            if dist < obs_radius + 0.5:
                return True
        return False
    
    @staticmethod
    def line_intersects_obstacle(p1, p2, obstacles):
        """Check if a line segment intersects any obstacle"""
        for obs in obstacles:
            if obs.get('moving', False):
                obs_pos = obs['pos']
                obs_radius = obs.get('radius', 1.0)
            else:
                obs_pos = obs['center'] if 'center' in obs else obs['pos']
                obs_radius = obs.get('radius', obs.get('size', 1.0))
            
            # Calculate distance from point to line segment
            p1_arr = np.array(p1)
            p2_arr = np.array(p2)
            obs_arr = np.array(obs_pos)
            
            line_vec = p2_arr - p1_arr
            point_vec = obs_arr - p1_arr
            line_len = np.linalg.norm(line_vec)
            
            if line_len < 0.01:
                return np.linalg.norm(point_vec) < obs_radius + 0.3
            
            line_unit = line_vec / line_len
            projection = np.dot(point_vec, line_unit)
            projection = np.clip(projection, 0, line_len)
            
            closest_point = p1_arr + line_unit * projection
            dist = np.linalg.norm(obs_arr - closest_point)
            
            if dist < obs_radius + 0.3:
                return True
        return False

    def on_click(self, event):
        if event.inaxes != self.ax:
            return
        
        x, y = event.xdata, event.ydata
        if not (0 <= x <= self.boundary_width and 0 <= y <= self.boundary_height):
            return
        
        x, y = round(x, 1), round(y, 1)
        
        if self.mode == 'add_job':
            # Don't add jobs on obstacles
            if not CollisionAvoidance.check_obstacle_collision((x, y), self.obstacles):
                self.add_job(x, y)
        elif self.mode == 'add_obstacle':
            self.add_obstacle_at(x, y)
        elif self.mode == 'set_start':
            if not CollisionAvoidance.check_obstacle_collision((x, y), self.obstacles):
                robot = self.robots[self.selected_robot]
                robot['start'] = (x, y)
                robot['pos'] = [x, y]
                robot['target_pos'] = [x, y]
                robot['kf'].set_initial_state([x, y])
                self.draw_all()
                self.update_status()
        elif self.mode == 'set_end':
            if not CollisionAvoidance.check_obstacle_collision((x, y), self.obstacles):
                self.robots[self.selected_robot]['end'] = (x, y)
                self.draw_all()
                self.update_status()
        elif self.mode == 'move_job':
            self.handle_move_job(x, y)
        elif self.mode == 'move_obstacle':
            self.handle_move_obstacle(x, y)
    
    def add_job(self, x, y):
        for job in self.jobs:
            if euclidean((x, y), job['pos']) < 1.0:
                return
        
        colors = ['#E74C3C', '#3498DB', '#2ECC71', '#F39C12', '#9B59B6',
                  '#1ABC9C', '#E67E22', '#16A085', '#C0392B', '#8E44AD']
        
        job = {
            'id': f'J{self.next_job_id}',
            'pos': (x, y),
            'color': colors[(self.next_job_id - 1) % len(colors)],
            'picked': False,
            'assigned_to': None
        }
        self.jobs.append(job)
        self.next_job_id += 1
        self.draw_all()
        self.update_status()
    
    def add_obstacle_at(self, x, y):
        """Add obstacle at clicked position"""
        for obs in self.obstacles:
            if euclidean((x, y), obs.get('center', obs.get('pos', (0, 0)))) < 2.0:
                return
        
        self.obstacles.append({
            'id': f'OBS{self.next_obs_id}',
            'type': 'static',
            'center': (x, y),
            'size': 1.2,
            'radius': 1.2,
            'shape': 'circle',
            'width': 2.4,
            'height': 2.4,
            'color': '#8B4513',
            'alpha': 0.7,
            'moving': False
        })
        self.next_obs_id += 1
        self.draw_all()
        self.update_status()
    
    def handle_move_job(self, x, y):
        if self.selected_job_idx is None:
            for i, job in enumerate(self.jobs):
                if euclidean((x, y), job['pos']) < 1.0:
                    self.selected_job_idx = i
                    self.instruction.set_text(f'Selected {job["id"]} | Click new location')
                    self.fig.canvas.draw_idle()
                    return
        else:
            if not CollisionAvoidance.check_obstacle_collision((x, y), self.obstacles):
                self.jobs[self.selected_job_idx]['pos'] = (x, y)
                self.selected_job_idx = None
                self.set_mode('add_job')
                self.draw_all()
                self.update_status()
    
    def handle_move_obstacle(self, x, y):
        if self.selected_obs_idx is None:
            for i, obs in enumerate(self.obstacles):
                obs_pos = obs.get('center', obs.get('pos', (0, 0)))
                if euclidean((x, y), obs_pos) < obs.get('radius', 1.0) + 0.5:
                    self.selected_obs_idx = i
                    self.instruction.set_text(f'Selected {obs["id"]} | Click new location')
                    self.fig.canvas.draw_idle()
                    return
        else:
            obs = self.obstacles[self.selected_obs_idx]
            if obs['moving']:
                obs['pos'] = [x, y]
            else:
                obs['center'] = (x, y)
            self.selected_obs_idx = None
            self.set_mode('add_job')
            self.draw_all()
            self.update_status()

    def assign_jobs_to_robots(self):
        for job in self.jobs:
            job['assigned_to'] = None
        
        if not self.jobs:
            return
        
        distances = []
        for j_idx, job in enumerate(self.jobs):
            for r_idx, robot in enumerate(self.robots):
                dist = euclidean(job['pos'], robot['start'])
                # Penalize if path crosses obstacles
                if CollisionAvoidance.line_intersects_obstacle(robot['start'], job['pos'], self.obstacles):
                    dist *= 2
                distances.append((dist, j_idx, r_idx))
        
        distances.sort()
        
        assigned_jobs = set()
        robot_counts = defaultdict(int)
        max_per_robot = max(1, len(self.jobs) // len(self.robots) + 1)
        
        for dist, j_idx, r_idx in distances:
            if j_idx not in assigned_jobs and robot_counts[r_idx] < max_per_robot:
                self.jobs[j_idx]['assigned_to'] = r_idx
                assigned_jobs.add(j_idx)
                robot_counts[r_idx] += 1
        
        for j_idx in range(len(self.jobs)):
            if j_idx not in assigned_jobs:
                min_robot = min(range(len(self.robots)), 
                              key=lambda r: robot_counts[r])
                self.jobs[j_idx]['assigned_to'] = min_robot
                robot_counts[min_robot] += 1
    
    def calculate_shortest_path(self, robot_idx):
        """Calculate shortest path avoiding obstacles"""
        robot = self.robots[robot_idx]
        assigned = [j for j in self.jobs if j['assigned_to'] == robot_idx]
        
        if not assigned:
            path = [robot['start'], robot['end']]
            robot['path'] = path
            robot['waypoints'] = path
            robot['total_distance'] = euclidean(robot['start'], robot['end'])
            robot['assigned_jobs'] = []
            return
        
        positions = [j['pos'] for j in assigned]
        
        min_dist = float('inf')
        best_order = None
        
        if len(positions) <= 8:
            for perm in itertools.permutations(range(len(positions))):
                dist = 0
                curr = robot['start']
                valid_path = True
                
                for idx in perm:
                    # Check if segment crosses obstacles
                    if CollisionAvoidance.line_intersects_obstacle(curr, positions[idx], self.obstacles):
                        valid_path = False
                        break
                    dist += euclidean(curr, positions[idx])
                    curr = positions[idx]
                
                if valid_path:
                    if not CollisionAvoidance.line_intersects_obstacle(curr, robot['end'], self.obstacles):
                        dist += euclidean(curr, robot['end'])
                        
                        if dist < min_dist:
                            min_dist = dist
                            best_order = list(perm)
        else:
            unvisited = list(range(len(positions)))
            curr = robot['start']
            best_order = []
            while unvisited:
                nearest = min(unvisited, key=lambda i: euclidean(curr, positions[i]))
                best_order.append(nearest)
                curr = positions[nearest]
                unvisited.remove(nearest)
            
            min_dist = euclidean(robot['start'], positions[best_order[0]])
            for i in range(len(best_order)-1):
                min_dist += euclidean(positions[best_order[i]], positions[best_order[i+1]])
            min_dist += euclidean(positions[best_order[-1]], robot['end'])
        
        if best_order:
            path = [robot['start']]
            robot['assigned_jobs'] = []
            for idx in best_order:
                path.append(positions[idx])
                robot['assigned_jobs'].append(assigned[idx])
            path.append(robot['end'])
        else:
            # Fallback path
            path = [robot['start'], robot['end']]
            robot['assigned_jobs'] = []
        
        robot['path'] = path
        robot['waypoints'] = path
        robot['total_distance'] = min_dist
    
    def calculate_all_paths(self, event):
        self.assign_jobs_to_robots()
        for i in range(len(self.robots)):
            self.calculate_shortest_path(i)
        self.draw_all()
        self.update_status()
    
    def check_collisions(self):
        collisions = []
        for i in range(len(self.robots)):
            for j in range(i + 1, len(self.robots)):
                dist = euclidean(self.robots[i]['pos'], self.robots[j]['pos'])
                if dist < 1.0:
                    collisions.append((i, j, dist))
        
        # Check obstacle collisions
        for i, robot in enumerate(self.robots):
            for obs in self.obstacles:
                obs_pos = obs.get('center', obs.get('pos', (0, 0)))
                dist = euclidean(robot['pos'], obs_pos)
                if dist < obs.get('radius', 1.0) + 0.5:
                    collisions.append((i, -1, dist))
        
        return collisions

    def update_dynamic_obstacles(self, dt):
        """Update positions of dynamic obstacles"""
        for obs in self.obstacles:
            if obs.get('moving', False):
                # Update position
                obs['pos'][0] += obs['velocity'][0] * dt
                obs['pos'][1] += obs['velocity'][1] * dt
                
                # Bounce off boundaries
                if obs['pos'][0] < 1 or obs['pos'][0] > self.boundary_width - 1:
                    obs['velocity'][0] *= -1
                if obs['pos'][1] < 1 or obs['pos'][1] > self.boundary_height - 1:
                    obs['velocity'][1] *= -1
                
                # Keep within bounds
                obs['pos'][0] = np.clip(obs['pos'][0], 1, self.boundary_width - 1)
                obs['pos'][1] = np.clip(obs['pos'][1], 1, self.boundary_height - 1)
    
    def update_animation(self, frame):
        if not self.animating:
            return []
        
        dt = 0.05
        self.simulation_time += dt
        
        # Update dynamic obstacles
        self.update_dynamic_obstacles(dt)
        
        all_done = True
        
        # Compute avoidance forces
        for robot in self.robots:
            robot['avoidance_force'] = [0, 0]
        
        if self.collision_avoidance_enabled:
            for i, robot in enumerate(self.robots):
                if robot['moving']:
                    other_robots = [self.robots[j] for j in range(len(self.robots)) if j != i]
                    avoidance = CollisionAvoidance.compute_avoidance_force(
                        robot['pos'], robot['velocity'], other_robots, self.obstacles,
                        safety_radius=robot['safety_radius']
                    )
                    robot['avoidance_force'] = avoidance.tolist()
        
        # Update each robot
        for robot in self.robots:
            if not robot['waypoints'] or robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                robot['moving'] = False
                robot['velocity'] = [0, 0]
                continue
            
            all_done = False
            robot['moving'] = True
            
            target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
            current = np.array(robot['pos'])
            
            direction = target - current
            distance = np.linalg.norm(direction)
            
            if distance < 0.5:
                robot['current_waypoint'] += 1
                
                r_idx = self.robots.index(robot)
                for job in self.jobs:
                    if (not job['picked'] and job['assigned_to'] == r_idx and 
                        euclidean(robot['pos'], job['pos']) < 1.0):
                        job['picked'] = True
                
                if robot['current_waypoint'] >= len(robot['waypoints']) - 1:
                    robot['moving'] = False
                    robot['velocity'] = [0, 0]
            else:
                direction_norm = direction / distance
                desired_velocity = direction_norm * robot['speed']
                
                if self.collision_avoidance_enabled:
                    avoidance = np.array(robot['avoidance_force'])
                    combined_velocity = desired_velocity + avoidance
                    
                    speed = np.linalg.norm(combined_velocity)
                    if speed > robot['speed'] * 1.5:
                        combined_velocity = combined_velocity / speed * robot['speed'] * 1.5
                    if speed < robot['speed'] * 0.3 and distance > 1.0:
                        combined_velocity = direction_norm * robot['speed'] * 0.5
                else:
                    combined_velocity = desired_velocity
                
                robot['kf'].state[2:] = combined_velocity
                robot['kf'].predict()
                
                noise = np.random.normal(0, 0.05, 2)
                measurement = current + combined_velocity * dt + noise
                
                estimated_pos = robot['kf'].update(measurement)
                
                # Boundary avoidance
                estimated_pos[0] = np.clip(estimated_pos[0], 0.5, self.boundary_width - 0.5)
                estimated_pos[1] = np.clip(estimated_pos[1], 0.5, self.boundary_height - 0.5)
                
                robot['pos'] = estimated_pos.tolist()
                robot['velocity'] = combined_velocity.tolist()
        
        if all_done:
            self.animating = False
            self.action_buttons['anim'].label.set_text('▶️ Start Animation')
            for robot in self.robots:
                robot['moving'] = False
                robot['velocity'] = [0, 0]
        
        self.draw_all()
        self.update_status()
        return []
    
    def draw_all(self):
        # Clear
        for artist in self.ax.collections[:]:
            artist.remove()

class ArUcoDetector:
    """Simulated ArUco marker detection"""
    def __init__(self, map_size=(30, 30)):
        self.map_size = map_size
        self.markers = {}  # Real marker positions
        self.detected_markers = {}  # Detected (with noise)
        self.camera_position = (15, 15)
        self.camera_range = 12
        self.detection_noise = 0.15
        
    def add_marker(self, marker_id, position, marker_type='job'):
        """Add a marker to the map"""
        self.markers[marker_id] = {
            'position': position,
            'type': marker_type,  # 'job', 'robot', 'obstacle'
            'last_detected': 0,
            'confidence': 1.0
        }
    
    def detect_markers(self, camera_pos, objects):
        """Simulate detecting markers from camera position"""
        detected = {}
        current_time = time.time()
        
        for obj_id, obj in objects.items():
            if isinstance(obj, dict) and 'pos' in obj:
                obj_pos = obj['pos']
            elif isinstance(obj, dict) and 'position' in obj:
                obj_pos = obj['position']
            else:
                continue
            
            dist = euclidean(camera_pos, obj_pos)
            
            if dist < self.camera_range:
                # Add detection noise
                noise_x = np.random.normal(0, self.detection_noise)
                noise_y = np.random.normal(0, self.detection_noise)
                
                detected_pos = (
                    obj_pos[0] + noise_x * dist/10,
                    obj_pos[1] + noise_y * dist/10
                )
                
                confidence = max(0, 1 - dist/self.camera_range - np.random.uniform(0, 0.1))
                
                detected[obj_id] = {
                    'position': detected_pos,
                    'true_position': obj_pos,
                    'confidence': confidence,
                    'timestamp': current_time
                }
        
        return detected

        for other_robot in self.robots:
            if other_robot['id'] != robot['id']:
                objects[other_robot['id']] = {'pos': other_robot['pos'], 'type': 'robot'}
        
        # Detect markers
        detected = self.aruco_detector.detect_markers(robot['pos'], objects)
        robot['detected_objects'] = detected
        
        # Update job detection status
        for job in self.jobs:
            job['detected'] = job['id'] in detected
    
    def real_time_loop(self):
        """Real-time simulation loop"""
        last_time = time.time()
        
        while True:
            current_time = time.time()
            dt = (current_time - last_time) * self.real_time_factor
            
            if self.running and not self.paused:
                self.simulation_time += dt
                
                # Update robots
                for robot in self.robots:
                    if robot['moving'] and robot['waypoints']:
                        self.move_robot(robot, dt)
                    
                    # Detect markers every 0.5 seconds
                    if self.simulation_time % 0.5 < dt:
                        self.detect_markers_for_robot(robot)
                    
                    # Drain battery
                    if robot['moving']:
                        robot['battery'] -= 0.1 * dt
                
                # Update FPS
                self.frame_count += 1
                if current_time - self.last_frame_time >= 1.0:
                    self.fps = self.frame_count
                    self.frame_count = 0
                    self.last_frame_time = current_time
            
            last_time = current_time
            time.sleep(0.03)  # ~30 FPS
    
    def move_robot(self, robot, dt):
        """Move robot towards next waypoint"""
        if robot['current_waypoint'] >= len(robot['waypoints']) - 1:
            robot['moving'] = False
            robot['status'] = 'completed'
            return
        
        target = np.array(robot['waypoints'][robot['current_waypoint'] + 1])
        current = np.array(robot['pos'])
        
        direction = target - current
        distance = np.linalg.norm(direction)
        
        if distance < 0.3:
            robot['current_waypoint'] += 1
            robot['status'] = 'picking' if distance < 0.5 else 'moving'
        else:
            direction_norm = direction / distance
            robot['velocity'] = (direction_norm * robot['speed']).tolist()
            robot['pos'] = (current + np.array(robot['velocity']) * dt).tolist()
            robot['status'] = 'moving'
    
    def update_animation(self, frame):

        self.time_text.set_text(f'Time: {self.simulation_time:.1f}s')
        
        if self.running and not self.paused:

        return []

                self.ax_main.text(job['pos'][0], job['pos'][1] + 0.5,

                                ha='center', fontsize=6, color='white',

                            fontweight='bold', color='white', zorder=6)

                vel_norm = np.array(robot['velocity'])
                vel_norm = vel_norm / np.linalg.norm(vel_norm) * 0.6
                arrow = FancyArrowPatch(
                    (robot['pos'][0], robot['pos'][1]),

                    color='white', linewidth=2.5, zorder=7
                )

    def draw_detection_panel(self):
        """Draw simulated camera view"""
        self.ax_detection.clear()
        self.ax_detection.set_facecolor('#1a1a2e')
        self.ax_detection.set_xlim(0, 640)
        self.ax_detection.set_ylim(0, 480)
        
        robot = self.robots[self.selected_robot] if self.robots else None
        
        if robot and robot.get('detected_objects'):
            detected = robot['detected_objects']
            
            # Draw detected objects as ArUco markers
            for i, (obj_id, obj_data) in enumerate(detected.items()):
                x = 100 + (i % 4) * 120
                y = 350 - (i // 4) * 120
                
                # Draw ArUco marker
                marker_size = 40
                rect = Rectangle((x - marker_size/2, y - marker_size/2),
                               marker_size, marker_size,
                               fill=False, color='#00ff00', linewidth=2)
                self.ax_detection.add_patch(rect)
                
                # Draw marker pattern
                for row in range(4):
                    for col in range(4):
                        if (row + col) % 2 == 0:
                            cell = Rectangle(
                                (x - marker_size/2 + col*10, y - marker_size/2 + row*10),
                                10, 10, fill=True, color='#00ff00', alpha=0.5
                            )
                            self.ax_detection.add_patch(cell)
                
                # Label
                self.ax_detection.text(x, y - 35, obj_id[:8],
                                      ha='center', fontsize=6, color='white')
                self.ax_detection.text(x, y + 35, f'Conf: {obj_data["confidence"]:.2f}',
                                      ha='center', fontsize=6, color='#00ff00')
        
        self.ax_detection.set_title(f'Camera: {robot["id"] if robot else "N/A"}', 
                                   fontsize=8, color='white')
        self.ax_detection.axis('off')
